const streams = {
  "متوسط": ["جميع المواد"],
  "ثانوي": Object.keys(CURRICULUM["ثانوي"]),
  "جامعي": Object.keys(CURRICULUM["جامعي"])
};

const YEARS = {
  "متوسط": ["الأولى متوسط", "الثانية متوسط", "الثالثة متوسط", "الرابعة متوسط"],
  "ثانوي": ["الأولى ثانوي", "الثانية ثانوي", "الثالثة ثانوي"],
  "جامعي": [
    "السنة الأولى ليسانس",
    "السنة الثانية ليسانس",
    "السنة الثالثة ليسانس",
    "السنة الأولى ماستر",
    "السنة الثانية ماستر",
    "أخرى"
  ]
};

export default function App() {
  const [stage, setStage] = useState("");
  const [year, setYear] = useState("");
  const [stream, setStream] = useState("");
  const [page, setPage] = useState("setup");
  const [lessons, setLessons] = useState([]);
  const [selected, setSelected] = useState(null);
  const [customSubjects, setCustomSubjects] = useState({});
  const [newSubject, setNewSubject] = useState("");
  const [search, setSearch] = useState("");
  const [premium, setPremium] = useState(false);

  useEffect(() => {
    loadData();
  }, []);

  async function loadData() {
    try {
      const saved = await AsyncStorage.getItem("study_lessons");
      const profile = await AsyncStorage.getItem("study_profile");
      const custom = await AsyncStorage.getItem("study_custom");

      if (saved) setLessons(JSON.parse(saved));
      if (custom) setCustomSubjects(JSON.parse(custom));

      if (profile) {
        const p = JSON.parse(profile);
        setStage(p.stage);
        setYear(p.year);
        setStream(p.stream);
        setPage("home");
      }
    } catch (e) {
      console.log("Load error", e);
    }
  }

  async function saveProfile() {
    if (!stage || !year || !stream) {
      Alert.alert("تنبيه", "يرجى اختيار الطور والسنة والتخصص");
      return;
    }

    await AsyncStorage.setItem(
      "study_profile",
      JSON.stringify({ stage, year, stream })
    );
    setPage("home");
  }

  function getSubjects() {
    const base = CURRICULUM[stage]?.[stream] || [];
    const custom = customSubjects[stage + stream] || [];
    return [...base, ...custom].filter(s =>
      s.name.toLowerCase().includes(search.toLowerCase())
    );
  }

  async function addSubject() {
    if (!newSubject.trim()) return;

    const key = stage + stream;
    const list = customSubjects[key] || [];

    const updated = {
      ...customSubjects,
      [key]: [
        ...list,
        subject(newSubject.trim(), "📘", "#2563EB")
      ]
    };

    setCustomSubjects(updated);
    await AsyncStorage.setItem(
      "study_custom",
      JSON.stringify(updated)
    );
    setNewSubject("");
  }

  async function takePhoto() {
    const permission =
      await ImagePicker.requestCameraPermissionsAsync();

    if (!permission.granted) {
      Alert.alert("إذن الكاميرا مطلوب");
      return;
    }

    const result = await ImagePicker.launchCameraAsync({
      mediaTypes: ["images"],
      allowsEditing: false,
      quality: 1,
      exif: false
    });

    if (!result.canceled) {
      const photo = result.assets[0].uri;
      saveLesson(photo);
    }
  }

  async function choosePhoto() {
    const permission =
      await ImagePicker.requestMediaLibraryPermissionsAsync();

    if (!permission.granted) {
      Alert.alert("إذن الصور مطلوب");
      return;
    }

    const result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ["images"],
      quality: 1
    });

    if (!result.canceled) {
      saveLesson(result.assets[0].uri);
    }
  }

  async function saveLesson(uri) {
    const item = {
      id: Date.now().toString(),
      uri,
      subject: selected.name,
      title: "درس جديد",
      date: new Date().toLocaleDateString()
    };

    const updated = [item, ...lessons];
    setLessons(updated);

    await AsyncStorage.setItem(
      "study_lessons",
      JSON.stringify(updated)
    );

    Alert.alert("نجاح", "تم حفظ الدرس");
  }

  async function deleteLesson(id) {
    Alert.alert("حذف الدرس", "هل تريد حذف هذا الدرس؟", [
      { text: "إلغاء" },
      {
        text: "حذف",
        onPress: async () => {
          const updated = lessons.filter(x => x.id !== id);
          setLessons(updated);
          await AsyncStorage.setItem(
            "study_lessons",
            JSON.stringify(updated)
          );
        }
      }
    ]);
  }

  const Button = ({ title, onPress, color = COLORS.primary }) => (
    <TouchableOpacity
      onPress={onPress}
      style={[styles.button, { backgroundColor: color }]}
    >
      <Text style={styles.buttonText}>{title}</Text>
    </TouchableOpacity>
  );

  const Option = ({ value, onPress, active }) => (
    <TouchableOpacity
      onPress={onPress}
      style={[
        styles.option,
        active && styles.activeOption
      ]}
    >
      <Text style={[
        styles.optionText,
        active && { color: COLORS.primary }
      ]}>{value}</Text>
    </TouchableOpacity>
  );

  const Header = ({ title }) => (
    <View style={styles.header}>
      <Text style={styles.headerTitle}>{title}</Text>
      <Text style={styles.brand}>Study Organizer</Text>
    </View>
  );

  if (page === "setup") {
    return (
      <SafeAreaView style={styles.screen}>
        <StatusBar barStyle="dark-content" />
        <ScrollView contentContainerStyle={styles.center}>
          <Text style={styles.logo}>📚</Text>
          <Text style={styles.appTitle}>Study Organizer</Text>
          <Text style={styles.designer}>Designed by Seddik Younes</Text>
          <Text style={styles.heading}>أنشئ ملفك الدراسي</Text>

          <Text style={styles.label}>اختر الطور الدراسي</Text>
          <View style={styles.wrap}>
            {["متوسط", "ثانوي", "جامعي"].map(x => (
              <Option key={x} value={x}
                active={stage === x}
                onPress={() => {
                  setStage(x);
                  setYear("");
                  setStream("");
                }}
              />
            ))}
          </View>

          {!!stage && <>
            <Text style={styles.label}>اختر السنة</Text>
            <View style={styles.wrap}>
              {(YEARS[stage] || []).map(x => (
                <Option key={x} value={x}
                  active={year === x}
                  onPress={() => setYear(x)}
                />
              ))}
            </View>
          </>}

          {!!year && <>
            <Text style={styles.label}>
              {stage === "جامعي" ? "اختر التخصص الجامعي" : "اختر الشعبة"}
            </Text>
            <View style={styles.wrap}>
              {(streams[stage] || []).map(x => (
                <Option key={x} value={x}
                  active={stream === x}
                  onPress={() => setStream(x)}
                />
              ))}
            </View>
          </>}

          <Button title="دخول إلى التطبيق" onPress={saveProfile} />
        </ScrollView>
      </SafeAreaView>
    );
  }

  if (page === "home") {
    return (
      <SafeAreaView style={styles.screen}>
        <Header title="موادي الدراسية" />
        <ScrollView contentContainerStyle={styles.content}>
          <View style={styles.profile}>
            <Text style={styles.profileTitle}>
              {stage} • {year}
            </Text>
            <Text style={styles.profileSub}>{stream}</Text>
            <TouchableOpacity onPress={() => setPage("setup")}>
              <Text style={styles.link}>تغيير الملف الدراسي</Text>
            </TouchableOpacity>
          </View>

          <TextInput
            style={styles.input}
            placeholder="ابحث عن مادة..."
            value={search}
            onChangeText={setSearch}
          />

          <View style={styles.grid}>
            {getSubjects().map((s, i) => (
              <TouchableOpacity
                key={i}
                style={styles.subjectCard}
                onPress={() => {
                  setSelected(s);
                  setPage("subject");
                }}
              >
                <View style={[
                  styles.subjectIcon,
                  { backgroundColor: s.color }
                ]}>
                  <Text style={styles.symbol}>{s.symbol}</Text>
                </View>
                <Text style={styles.subjectName}>{s.name}</Text>
                <Text style={styles.small}>
                  {lessons.filter(l => l.subject === s.name).length} درس
                </Text>
              </TouchableOpacity>
            ))}
          </View>

          <Text style={styles.label}>إضافة مادة مخصصة</Text>
          <TextInput
            style={styles.input}
            placeholder="اسم المادة الجديدة"
            value={newSubject}
            onChangeText={setNewSubject}
          />
          <Button title="إضافة المادة" onPress={addSubject} />

          <Button title="إعدادات التطبيق" color="#475569"
            onPress={() => setPage("settings")} />
        </ScrollView>
      </SafeAreaView>
    );
  }

  if (page === "subject") {
    const subjectLessons = lessons.filter(
      x => x.subject === selected?.name
    );

    return (
      <SafeAreaView style={styles.screen}>
        <Header title={selected?.name || "المادة"} />
        <ScrollView contentContainerStyle={styles.content}>
          <Button title="التقاط صورة للدرس 📷" onPress={takePhoto} />
          <Button title="اختيار صورة من المعرض" color="#059669"
            onPress={choosePhoto} />

          <Text style={styles.heading}>دروسي</Text>

          {subjectLessons.map(item => (
            <View key={item.id} style={styles.lesson}>
              <Image source={{ uri: item.uri }}
                style={styles.lessonImage} />
              <View style={{ flex: 1 }}>
                <Text style={styles.subjectName}>{item.title}</Text>
                <Text style={styles.small}>{item.date}</Text>
                <TouchableOpacity onPress={() => deleteLesson(item.id)}>
                  <Text style={{ color: "#DC2626", marginTop: 8 }}>
                    حذف الدرس
                  </Text>
                </TouchableOpacity>
              </View>
            </View>
          ))}

          {subjectLessons.length === 0 &&
            <Text style={styles.empty}>لا توجد دروس بعد</Text>}

          <Button title="العودة للمواد" color="#475569"
            onPress={() => setPage("home")} />
        </ScrollView>
      </SafeAreaView>
    );
  }

  if (page === "settings") {
    return (
      <SafeAreaView style={styles.screen}>
        <Header title="الإعدادات" />
        <View style={styles.content}>
          <Text style={styles.heading}>Study Organizer</Text>
          <Text style={styles.profileSub}>
            المصمم: Seddik Younes
          </Text>

          <Text style={styles.label}>تحسين وضوح الصور</Text>
          <Text style={styles.small}>
            ميزة اختيارية مدفوعة. تفعيلها يتطلب دمج خدمة
            تحسين الصور ونظام دفع حقيقي.
          </Text>

          <Button title="معاينة ميزة Premium"
            onPress={() => {
              Alert.alert(
                "Premium",
                "ميزة تحسين الصور اختيارية وستكون متاحة بعد ربط نظام الدفع."
              );
            }}
            color="#7C3AED"
          />

          <Button title="العودة للرئيسية" color="#475569"
            onPress={() => setPage("home")} />
        </View>
      </SafeAreaView>
    );
  }

  return null;
}
