let cameraStream = null;
let lastPhoto = null;


/* NAVIGATION */

function showSection(id,btn){

  document.querySelectorAll('.section').forEach(section=>{
    section.classList.remove('active');
  });

  const section=document.getElementById(id);

  if(section){
    section.classList.add('active');
    window.scrollTo({top:0,behavior:'smooth'});
  }

  if(btn){
    document.querySelectorAll('nav button').forEach(b=>{
      b.classList.remove('active');
    });

    btn.classList.add('active');
  }
}


/* CAMERA */

async function openCamera(){

  try{

    cameraStream=await navigator.mediaDevices.getUserMedia({
      video:{
        facingMode:"environment"
      },
      audio:false
    });

    document.getElementById("camera").srcObject=cameraStream;

  }catch(error){

    alert(
      "Impossible d'accéder à la caméra. Vérifiez les permissions du navigateur."
    );

  }
}


function takePhoto(){

  const video=document.getElementById("camera");
  const canvas=document.getElementById("canvas");

  if(!video.srcObject){

    alert("Ouvrez d'abord la caméra.");

    return;
  }

  canvas.width=video.videoWidth;
  canvas.height=video.videoHeight;

  const ctx=canvas.getContext("2d");

  ctx.drawImage(video,0,0);

  lastPhoto=canvas.toDataURL("image/jpeg");

  alert("Photo capturée avec succès.");

}

/* ENREGISTREMENT */

function saveDiagnosis(){

  const vehicle=document.getElementById("diagVehicle").value || "Véhicule non renseigné";
  const problem=document.getElementById("problem").value;

  const history=JSON.parse(
    localStorage.getItem("mecaHistory") || "[]"
  );

  history.unshift({

    date:new Date().toLocaleString("fr-FR"),

    vehicle:vehicle,

    problem:problem

  });

  localStorage.setItem(
    "mecaHistory",
    JSON.stringify(history)
  );

  alert("Diagnostic enregistré.");

  loadHistory();

}


/* HISTORIQUE */

function loadHistory(){

  const history=JSON.parse(
    localStorage.getItem("mecaHistory") || "[]"
  );

  const list=document.getElementById("historyList");

  if(history.length===0){

    list.innerHTML=`
      <div class="result">
        Aucun diagnostic enregistré pour le moment.
      </div>
    `;

    return;
  }

  list.innerHTML=history.map((item,index)=>`

    <div class="historyItem">

      <strong>${item.vehicle}</strong>

      <p>Problème : ${item.problem}</p>

      <small>${item.date}</small>

    </div>

  `).join("");

}

      Une lecture des codes OBD peut être nécessaire.
    `;

  }else if(light==="Voyant huile"){

    text=`
      Ce voyant peut indiquer un problème de pression ou de niveau
      d'huile. Arrêtez le moteur si le voyant reste allumé et
      effectuez les contrôles appropriés.
    `;

  }else if(light==="Voyant batterie"){

    text=`
      Ce voyant peut être lié au système de charge :
      alternateur, courroie, batterie ou circuit électrique.
    `;

  }else{

    text=`
      Le voyant indique qu'un système du véhicule a signalé
      une anomalie ou une condition nécessitant une vérification.
      Consultez la documentation du véhicule et effectuez
      un diagnostic adapté.
    `;

  }

  document.getElementById("warningResult").innerHTML=`

    <div class="result">

      <span class="badge">ANALYSE</span>

      <h3>${light}</h3>

      <p>${text}</p>

      <div class="warning">
        ⚠️ L'interprétation exacte dépend du véhicule et de son système
        électronique. Utilisez un outil de diagnostic lorsque nécessaire.
      </div>

    </div>

  `;

}


/* AUDIO */

async function startAudio(){

  try{

    const stream=await navigator.mediaDevices.getUserMedia({
      audio:true
    });

    alert(
      "Microphone activé. Enregistrez maintenant le bruit du véhicule."
    );

    setTimeout(()=>{

      stream.getTracks().forEach(track=>track.stop());

      alert(
        "Enregistrement terminé. L'analyse sonore IA pourra être connectée dans la prochaine version."
      );

    },5000);

  }catch(error){

    alert(
      "Accès au microphone refusé ou non disponible."
    );

  }

}
