/* ---------- Images ---------- */
function reduireImage(fichier, max){
  return new Promise((ok, ko) => {
    const l = new FileReader(); l.onerror = ko;
    l.onload = () => {
      const img = new Image(); img.onerror = ko;
      img.onload = () => {
        const e = Math.min(1, max / Math.max(img.width, img.height));
        const c = document.createElement("canvas"); c.width = Math.round(img.width * e); c.height = Math.round(img.height * e);
        c.getContext("2d").drawImage(img, 0, 0, c.width, c.height);
        ok(c.toDataURL("image/jpeg", 0.7));
      };
      img.src = l.result;
    };
    l.readAsDataURL(fichier);
  });
}

}

function afficherPosts(posts){
  const zone = $("#liste-posts"); zone.innerHTML = "";
  if (!posts.length) { zone.appendChild(el("div", { className: "vide", textContent: "Aucune publication. Soyez le premier !" })); return; }
  posts.forEach(p => {
    const dejaAime = (p.aimes || []).includes(user.uid);
    const date = p.date ? p.date.toDate().toLocaleString("fr-FR") : "à l'instant";
    const av = el("div", { className: "avatar" }); pastille(av, p.auteur, p.photo);
    const carte = el("div", { className: "carte" },
      el("div", { className: "auteur" }, av, el("div", {},
        el("b", { textContent: p.auteur }), el("span", { className: "tag", textContent: p.type }),
        el("div", { className: "meta", textContent: (p.role === "Enseignant" ? "🧑‍🏫 " : "🎓 ") + p.role + (p.pays ? "  🌍 " + p.pays : "") + "  🕒 " + date }))),
      el("div", { textContent: p.texte }));
    if (p.image) carte.appendChild(el("img", { src: p.image, alt: "Image publiée", className: "media" }));
    const actions = el("div", { className: "actions" },
      el("button", { className: "petit", textContent: (dejaAime ? "💙" : "👍") + " J'aime (" + (p.aimes || []).length + ")",
        onclick: () => updateDoc(doc(db, "posts", p.id), { aimes: dejaAime ? arrayRemove(user.uid) : arrayUnion(user.uid) }) }),
      el("button", { className: "petit", textContent: "💬 Commenter", onclick: () => {
        const t = prompt("Votre commentaire :"); if (t && t.trim())
          updateDoc(doc(db, "posts", p.id), { commentaires: arrayUnion({ auteur: profil.nom || "Anonyme", texte: t.trim() }) });
      } }),
      el("button", { className: "petit", textContent: "📤 Partager", onclick: () => partagerTexte(p.auteur + " sur Campus : " + p.texte) }));
    if (p.uid === user.uid) actions.appendChild(el("button", { className: "petit", textContent: "🗑️ Supprimer",
      onclick: () => { if (confirm("Supprimer cette publication ?")) deleteDoc(doc(db, "posts", p.id)); } }));
    carte.appendChild(actions);
    (p.commentaires || []).forEach(c => carte.appendChild(el("div", { className: "comm", textContent: c.auteur + " : " + c.texte })));
    zone.appendChild(carte);
  });
}

function afficherAmis(){
  if (!user) return;
  const q = $("#a-recherche").value.trim().toLowerCase();
  const vide = (z, t) => z.appendChild(el("div", { className: "vide", textContent: t }));
  const demandes = $("#liste-demandes"), amisZ = $("#liste-amis"), pers = $("#liste-personnes");
  [demandes, amisZ, pers].forEach(z => z.innerHTML = "");

  const autreDe = a => a.membres.find(m => m !== user.uid);
  const recues = amities.filter(a => a.statut === "attente" && a.vers === user.uid);
  const mesAmis = amities.filter(a => a.statut === "ami");

  recues.forEach(a => demandes.appendChild(carteAmi(a.de, el("div", {},
    bt("✅ Accepter", () => updateDoc(doc(db, "amities", a.id), { statut: "ami" })),
    bt("Refuser", () => deleteDoc(doc(db, "amities", a.id)), "second")))));
  if (!recues.length) vide(demandes, "Aucune demande.");

  $("#nb-amis").textContent = mesAmis.length;
  mesAmis.forEach(a => amisZ.appendChild(carteAmi(autreDe(a), el("div", {},
    bt("📞 Appeler", () => appeler(a)),
    bt("Retirer", () => { if (confirm("Retirer cet ami ?")) deleteDoc(doc(db, "amities", a.id)); }, "second")))));
  if (!mesAmis.length) vide(amisZ, "Aucun ami pour le moment.");

  let n = 0;
  Object.keys(personnes).forEach(uid => {
    const p = personnes[uid];
    if (uid === user.uid || !p.nom) return;
    if (q && !(p.nom + " " + (p.fil || "") + " " + (p.pays || "")).toLowerCase().includes(q)) return;
    const rel = amities.find(a => a.membres.includes(uid));
    if (rel && rel.statut === "ami") return;
    let b;
    if (!rel) b = bt("➕ Ajouter", () => setDoc(doc(db, "amities", user.uid + "_" + uid),
      { de: user.uid, vers: uid, membres: [user.uid, uid], statut: "attente" }));
    else if (rel.de === user.uid) b = bt("⏳ Demande envoyée (annuler)", () => deleteDoc(doc(db, "amities", rel.id)), "second");
    else b = bt("✅ Accepter", () => updateDoc(doc(db, "amities", rel.id), { statut: "ami" }));
    pers.appendChild(carteAmi(uid, b)); n++;
  });
  if (!n) vide(pers, "Personne à afficher.");
}

function entrerSalle(code){
  code = code.toLowerCase().replace(/[^a-z0-9-]/g, "-").replace(/-+/g, "-").slice(0, 60);
  if (!code) { alert("Entrez ou générez un code de salle."); return; }
  lienAppel = "https://meet.jit.si/campus-" + code;
  $("#lien-appel").textContent = lienAppel;
  $("#cadre").src = lienAppel + "#userInfo.displayName=" + encodeURIComponent('"' + (profil.nom || "Participant") + '"') + "&config.prejoinPageEnabled=false";
  $("#entree-cours").style.display = "none"; $("#salle-video").style.display = "block";
  ouvrir("cours");
}

// Appeler un ami : l'ami voit une bannière « vous appelle » sur son écran
async function appeler(a){
  const salle = "ami-" + Math.random().toString(36).slice(2, 8);
  try { await updateDoc(doc(db, "amities", a.id), { appel: { salle, par: user.uid, nom: profil.nom || "Un ami", ts: Date.now() } }); }
  catch (e) { alert("Appel impossible : " + (e.code || e.message)); return; }
  entrerSalle(salle);
}
function verifierAppel(){
  const b = $("#bandeau");
  const a = amities.find(x => x.appel && x.appel.par !== user.uid && Date.now() - x.appel.ts < 5 * 60 * 1000);
  if (!a) { b.style.display = "none"; return; }
  b.innerHTML = ""; b.style.display = "block";
  b.append("📞 " + a.appel.nom + " vous appelle ", el("button", { textContent: "Rejoindre", onclick: () => { b.style.display = "none"; entrerSalle(a.appel.salle); } }));
}

$("#btn-ressource").onclick = async () => {
  const titre = $("#b-titre").value.trim(); if (!titre) return;
  let lien = $("#b-lien").value.trim(); if (lien && !/^https?:\/\//i.test(lien)) lien = "";
  try {
    await addDoc(collection(db, "biblio"), { uid: user.uid, par: profil.nom || "Anonyme", titre, matiere: $("#b-matiere").value.trim(), lien, date: serverTimestamp() });
    $("#b-titre").value = $("#b-matiere").value = $("#b-lien").value = "";
  } catch (e) { alert("Ajout impossible : " + (e.code || e.message)); }
};
$("#b-recherche").oninput = afficherBiblio;
function afficherBiblio(){
  const zone = $("#liste-biblio"); zone.innerHTML = "";
  const q = $("#b-recherche").value.trim().toLowerCase();
  const liste = ressources.filter(r => (r.titre + " " + r.matiere).toLowerCase().includes(q));
  if (!liste.length) { zone.appendChild(el("div", { className: "vide", textContent: "Aucune ressource." })); return; }
  liste.forEach(r => {
    const nom = el("b", {}, r.lien ? el("a", { href: r.lien, target: "_blank", rel: "noopener", textContent: r.titre }) : r.titre);
    if (r.matiere) nom.appendChild(el("span", { className: "tag", textContent: r.matiere }));
    const l = el("div", { className: "ligne-liste" }, el("div", {}, nom, el("div", { className: "meta", textContent: "Ajouté par " + r.par })));
    if (r.uid === user.uid) l.appendChild(el("button", { className: "petit", textContent: "🗑️", "aria-label": "Supprimer", onclick: () => deleteDoc(doc(db, "biblio", r.id)) }));
    zone.appendChild(l);
  });
}
