
import React, { useState } from 'react';
import {
  View, Text, TextInput, ScrollView,
  TouchableOpacity, StyleSheet
} from 'react-native';

const initialPosts = [
  {id:1, user:'Alex Morgan', handle:'@alex',
   text:'Welcome to Craving! A new place to connect and share.', likes:24},
  {id:2, user:'Sarah', handle:'@sarah',
   text:'Every day is a new opportunity to discover something amazing.', likes:12},
];

export default function App() {
  const [tab, setTab] = useState('Home');
  const [posts, setPosts] = useState(initialPosts);
  const [liked, setLiked] = useState([]);
  const [draft, setDraft] = useState('');

  const addPost = () => {
    if (!draft.trim()) return;
    setPosts([{
      id: Date.now(),
      user:'You',
      handle:'@you',
      text:draft,
      likes:0
    }, ...posts]);
    setDraft('');
    setTab('Home');
  };

  const toggleLike = (id) => {
    setLiked(old =>
      old.includes(id)
        ? old.filter(x => x !== id)
        : [...old, id]
    );
  };

  return (
    <View style={s.container}>
      <View style={s.header}>
        <Text style={s.logo}>Craving.</Text>
        <Text style={s.sub}>Connect. Share. Discover.</Text>
      </View>

      {tab === 'Home' && (
        <ScrollView style={s.feed}>
          <Text style={s.heading}>Your Feed</Text>
          {posts.map(p => (
            <View key={p.id} style={s.post}>
              <Text style={s.user}>{p.user}</Text>
              <Text style={s.handle}>{p.handle}</Text>
              <Text style={s.message}>{p.text}</Text>
              <TouchableOpacity onPress={() => toggleLike(p.id)}>
                <Text style={s.like}>
                  {liked.includes(p.id) ? '❤️' : '♡'}
                  {' '} {p.likes + (liked.includes(p.id) ? 1 : 0)} Likes
                </Text>
              </TouchableOpacity>
            </View>
          ))}
        </ScrollView>
      )}

      {tab === 'Create' && (
        <View style={s.screen}>
          <Text style={s.heading}>Create a Post</Text>
          <TextInput
            style={s.input}
            multiline
            placeholder="What's on your mind?"
            value={draft}
            onChangeText={setDraft}
          />
          <TouchableOpacity style={s.button} onPress={addPost}>
            <Text style={s.buttonText}>Publish Post</Text>
          </TouchableOpacity>
        </View>
      )}

      {tab === 'Search' && (
        <View style={s.screen}>
          <Text style={s.heading}>Discover</Text>
          <TextInput style={s.input}
            placeholder="Search Craving..." />
        </View>
      )}

      {tab === 'Activity' && (
        <View style={s.screen}>
          <Text style={s.heading}>Activity</Text>
          <Text>You're all caught up!</Text>
        </View>
      )}

      {tab === 'Profile' && (
        <View style={s.screen}>
          <Text style={s.heading}>Your Profile</Text>
          <Text style={s.user}>Craving User</Text>
          <Text>@you</Text>
          <Text>{posts.filter(p => p.handle === '@you').length} Posts</Text>
        </View>
      )}

      <View style={s.nav}>
        {['Home','Search','Create','Activity','Profile'].map(t => (
          <TouchableOpacity key={t} onPress={() => setTab(t)}>
            <Text style={[s.navText, tab === t && s.active]}>
              {t === 'Home' ? '⌂' :
               t === 'Search' ? '⌕' :
               t === 'Create' ? '+' :
               t === 'Activity' ? '♡' : '◎'}
            </Text>
            <Text style={[s.navLabel, tab === t && s.active]}>
              {t}
            </Text>
          </TouchableOpacity>
        ))}
      </View>
    </View>
  );
}

const s = StyleSheet.create({
  container:{flex:1,backgroundColor:'#F8F9FC',paddingTop:45},
  header:{padding:20,backgroundColor:'#fff'},
  logo:{fontSize:30,fontWeight:'bold',color:'#222'},
  sub:{color:'#888',fontSize:12},
  feed:{padding:16},
  heading:{fontSize:23,fontWeight:'bold',marginBottom:18,color:'#222'},
  post:{backgroundColor:'#fff',padding:18,borderRadius:15,marginBottom:14},
  user:{fontSize:16,fontWeight:'bold',color:'#222'},
  handle:{color:'#999',marginBottom:12},
  message:{fontSize:15,lineHeight:23,color:'#444'},
  like:{marginTop:16,color:'#555',fontWeight:'600'},
  screen:{flex:1,padding:20},
  input:{backgroundColor:'#fff',padding:16,borderRadius:12,
    minHeight:55,textAlignVertical:'top',marginBottom:16},
  button:{backgroundColor:'#222',padding:16,borderRadius:12},
  buttonText:{color:'#fff',textAlign:'center',fontWeight:'bold'},
  nav:{flexDirection:'row',justifyContent:'space-around',
    padding:13,backgroundColor:'#fff',borderTopWidth:1,borderColor:'#eee'},
  navText:{fontSize:23,textAlign:'center',color:'#888'},
  navLabel:{fontSize:10,color:'#888',textAlign:'center'},
  active:{color:'#2563EB',fontWeight:'bold'}
});
