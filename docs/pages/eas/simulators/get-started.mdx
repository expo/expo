package com.najm.keyboard

import android.inputmethodservice.InputMethodService
import android.inputmethodservice.Keyboard
import android.inputmethodservice.KeyboardView
import android.view.KeyEvent
import android.view.View
import android.view.inputmethod.EditorInfo
import com.najm.keyboard.sync.LocalLexiconStore
import com.najm.keyboard.sync.PrivacyGuard
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.launch

class NajmKeyboardService : InputMethodService(), KeyboardView.OnKeyboardActionListener {

    private val serviceScope = CoroutineScope(SupervisorJob() + Dispatchers.IO)
    private lateinit var lexiconStore: LocalLexiconStore
    private var keyboardView: KeyboardView? = null

    private val arabicKeyboard by lazy { Keyboard(this, R.xml.keyboard_arabic) }
    private val symbolsKeyboard by lazy { Keyboard(this, R.xml.keyboard_symbols) }

    override fun onCreate() {
        super.onCreate()
        lexiconStore = LocalLexiconStore(this)
    }

    override fun onDestroy() {
        serviceScope.cancel()
        super.onDestroy()
    }

    override fun onCreateInputView(): View {
        keyboardView = KeyboardView(this, null).apply {
            keyboard = arabicKeyboard
            setOnKeyboardActionListener(this@NajmKeyboardService)
        }
        return keyboardView!!
    }

    override fun onStartInputView(info: EditorInfo?, restarting: Boolean) {
        super.onStartInputView(info, restarting)
        keyboardView?.keyboard = arabicKeyboard
    }

    override fun onKey(primaryCode: Int, keyCodes: IntArray?) {
        val ic = currentInputConnection ?: return
        when (primaryCode) {
            Keyboard.KEYCODE_DELETE -> {
                if (ic.getSelectedText(0) != null) {
                    ic.commitText("", 1)
                } else {
                    sendDownUpKeyEvents(KeyEvent.KEYCODE_DEL)
                }
            }
            Keyboard.KEYCODE_MODE_CHANGE -> {
                keyboardView?.keyboard = symbolsKeyboard
            }
            KEYCODE_ARABIC_MODE -> {
                keyboardView?.keyboard = arabicKeyboard
            }
            KEYCODE_SHIFT -> {
                keyboardView?.setShifted(keyboardView?.isShifted?.not() ?: true)
            }
            Keyboard.KEYCODE_DONE -> {
                sendDownUpKeyEvents(KeyEvent.KEYCODE_ENTER)
            }
            else -> {
                if (primaryCode <= 0) return
                var text = primaryCode.toChar().toString()
                if (keyboardView?.isShifted == true) {
                    text = text.uppercase()
                }
                ic.commitText(text, 1)
                if (text == " ") recordLastWord()
            }
        }
    }

    /** تسجيل الكلمة الأخيرة في القاموس الشخصي (O(1) على Room) مع احترام وضع الخصوصية. */
    private fun recordLastWord() {
        val editorInfo = currentInputEditorInfo ?: return
        val isPrivate = PrivacyGuard.isPrivateField(editorInfo.imeOptions, editorInfo.inputType)
        if (isPrivate) return
        val before = currentInputConnection?.getTextBeforeCursor(64, 0) ?: return
        val word = before.trim().split(Regex("\\s+")).lastOrNull() ?: return
        if (word.isBlank() || word.length > 120) return
        serviceScope.launch { lexiconStore.recordWordUsage(word, isPrivateMode = false) }
    }

    override fun onPress(primaryCode: Int) {}
    override fun onRelease(primaryCode: Int) {}
    override fun onText(text: CharSequence?) {
        currentInputConnection?.commitText(text, 1)
    }
    override fun swipeLeft() {}
    override fun swipeRight() {}
    override fun swipeDown() {}
    override fun swipeUp() {}

    private companion object {
        const val KEYCODE_SHIFT = -1
        const val KEYCODE_ARABIC_MODE = -3
    }
}

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isVerifiedOwner(userId) {
      return isSignedIn()
        && request.auth.token.email_verified == true
        && request.auth.uid == userId;
    }

    function isValidUserProfile(data, userId) {
      return data.keys().hasAll(['uid', 'email', 'displayName', 'wordCount', 'lastSyncedAt'])
        && data.keys().hasOnly(['uid', 'email', 'displayName', 'wordCount', 'lastSyncedAt'])
        && data.uid == userId
        && data.email is string
        && data.email.size() > 0
        && data.email.size() <= 320
        && data.displayName is string
        && data.displayName.size() <= 120
        && data.wordCount is int
        && data.wordCount >= 0
        && data.wordCount <= 500000
        && data.lastSyncedAt is timestamp;
    }

    function isValidLexiconWord(data, userId, wordId) {
      return wordId.matches('^[a-f0-9]{64}$')
        && data.keys().hasAll(['ownerId', 'word', 'count', 'updatedAt', 'deleted'])
        && data.keys().hasOnly(['ownerId', 'word', 'count', 'updatedAt', 'deleted'])
        && data.ownerId == userId
        && data.word is string
        && data.word.size() >= 1
        && data.word.size() <= 120
        && !data.word.matches('^\\s*$')
        && data.count is int
        && data.count >= 1
        && data.count <= 10000000
        && data.updatedAt is timestamp
        && data.deleted is bool;
    }

    match /users/{userId} {
      allow read, delete: if isVerifiedOwner(userId);
      allow create, update: if isVerifiedOwner(userId)
        && isValidUserProfile(request.resource.data, userId);

      match /lexicon_words/{wordId} {
        allow read, delete: if isVerifiedOwner(userId);
        allow create, update: if isVerifiedOwner(userId)
          && isValidLexiconWord(request.resource.data, userId, wordId);
      }
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}

# كيبورد نجم

لوحة مفاتيح عربية لأندرويد مع مزامنة سحابية للقاموس الشخصي.

## خطوات التشغيل

1. افتح مجلد المشروع في **Android Studio** (سيجلب Gradle تلقائياً عبر `gradle-wrapper.properties`).
2. ضع صورتك كأيقونة عالية الجودة: **File > New > Image Asset > Launcher Icons** واختر `app/src/main/res/drawable-nodpi/app_icon.jpg` كمصدر (أو اترك الأيقونة الحالية).
3. للمزامنة السحابية (اختياري):
   - أنشئ مشروعاً في console.firebase.google.com وأضف تطبيق أندرويد بالمعرّف `com.najm.keyboard`.
   - أضف بصمات SHA-1 وSHA-256 (من `./gradlew signingReport`) في إعدادات التطبيق بـ Firebase.
   - نزّل `google-services.json` وضعه في مجلد `app/`.
