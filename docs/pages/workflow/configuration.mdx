
import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  ScrollView,
  StyleSheet,
  Alert
} from 'react-native';

export default function App() {
  const [role, setRole] = useState('customer');
  const [pickup, setPickup] = useState('');
  const [destination, setDestination] = useState('');
  const [fee, setFee] = useState('');
  const [details, setDetails] = useState('');
  const [orders, setOrders] = useState([]);

  const createOrder = () => {
    if (!pickup.trim() || !destination.trim() || !fee.trim()) {
      Alert.alert('تنبيه', 'يرجى ملء الحقول المطلوبة');
      return;
    }

    const amount = Number(fee);

    if (!Number.isFinite(amount) || amount <= 0) {
      Alert.alert('تنبيه', 'أدخل أجرة صحيحة');
      return;
    }

    const newOrder = {
      id: Date.now().toString(),
      pickup,
      destination,
      fee: amount,
      details,
      status: 'pending'
    };

    setOrders(previous => [newOrder, ...previous]);
    setPickup('');
    setDestination('');
    setFee('');
    setDetails('');

    Alert.alert('تم بنجاح', 'تم إنشاء طلبك');
  };

  const acceptOrder = (id) => {
    setOrders(previous =>
      previous.map(order =>
        order.id === id
          ? { ...order, status: 'accepted' }
          : order
      )
    );
  };

  const finishOrder = (id) => {
    setOrders(previous =>
      previous.map(order =>
        order.id === id
          ? { ...order, status: 'completed' }
          : order
      )
    );
  };

  const statusName = (status) => {
    if (status === 'pending') return 'بانتظار مندوب';
    if (status === 'accepted') return 'تم قبول الطلب';
    return 'تم إكمال التوصيل';
  };

  return (
    <View style={styles.container}>
      <View style={styles.header}>
        <Text style={styles.logo}>وصّلني 🇮🇶</Text>
        <Text style={styles.subtitle}>خدمة التوصيل في العراق</Text>
      </View>

      <View style={styles.tabs}>
        <TouchableOpacity
          style={[styles.tab, role === 'customer' && styles.activeTab]}
          onPress={() => setRole('customer')}
        >
          <Text style={styles.tabText}>الزبون</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={[styles.tab, role === 'driver' && styles.activeTab]}
          onPress={() => setRole('driver')}
        >
          <Text style={styles.tabText}>المندوب</Text>
        </TouchableOpacity>
      </View>

      <ScrollView contentContainerStyle={styles.content}>
        <Text style={styles.location}>📍 صلاح الدين - بيجي</Text>

        {role === 'customer' && (
          <View>
            <Text style={styles.heading}>اطلب مندوباً الآن</Text>

            <TextInput
              style={styles.input}
              placeholder="مكان الاستلام"
              value={pickup}
              onChangeText={setPickup}
            />

            <TextInput
              style={styles.input}
              placeholder="مكان التسليم"
              value={destination}
              onChangeText={setDestination}
            />

            <TextInput
              style={styles.input}
              placeholder="تفاصيل الطلب"
              value={details}
              onChangeText={setDetails}
            />

            <TextInput
              style={styles.input}
              placeholder="أجرة التوصيل بالدينار العراقي"
              keyboardType="numeric"
              value={fee}
              onChangeText={setFee}
            />

            <TouchableOpacity
              style={styles.button}
              onPress={createOrder}
            >
              <Text style={styles.buttonText}>إنشاء طلب التوصيل</Text>
            </TouchableOpacity>
          </View>
        )}

        <Text style={styles.heading}>
          {role === 'driver' ? 'الطلبات المتاحة' : 'طلباتي'}
        </Text>

        {orders.length === 0 && (
          <Text style={styles.empty}>لا توجد طلبات حالياً</Text>
        )}

        {orders.map(order => (
          <View key={order.id} style={styles.card}>
            <Text style={styles.cardTitle}>
              طلب رقم {order.id.slice(-5)}
            </Text>

            <Text style={styles.info}>الاستلام: {order.pickup}</Text>
            <Text style={styles.info}>التسليم: {order.destination}</Text>
            <Text style={styles.info}>
              التفاصيل: {order.details || 'لا توجد'}
            </Text>
            <Text style={styles.price}>
              الأجرة: {order.fee.toLocaleString('en-US')} د.ع
            </Text>
            <Text style={styles.status}>
              {statusName(order.status)}
            </Text>

            {role === 'driver' && order.status === 'pending' && (
              <TouchableOpacity
                style={styles.button}
                onPress={() => acceptOrder(order.id)}
              >
                <Text style={styles.buttonText}>قبول الطلب</Text>
              </TouchableOpacity>
            )}

            {role === 'driver' && order.status === 'accepted' && (
              <TouchableOpacity
                style={styles.finishButton}
                onPress={() => finishOrder(order.id)}
              >
                <Text style={styles.buttonText}>إنهاء التوصيل</Text>
              </TouchableOpacity>
            )}
          </View>
        ))}
      </ScrollView>
    </View>
  );
}
