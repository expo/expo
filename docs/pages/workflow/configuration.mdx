
import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  ScrollView,
  StyleSheet,
  Alert,
  StatusBar
} from 'react-native';

export default function App() {
  const [role, setRole] = useState('customer');

  const [pickup, setPickup] = useState('');
  const [destination, setDestination] = useState('');
  const [details, setDetails] = useState('');
  const [fee, setFee] = useState('');

  const [orders, setOrders] = useState([]);

  const createOrder = () => {
    if (!pickup || !destination || !fee) {
      Alert.alert('تنبيه', 'يرجى ملء جميع الحقول المطلوبة');
      return;
    }

    if (Number(fee) <= 0 || !Number.isFinite(Number(fee))) {
      Alert.alert('تنبيه', 'أدخل أجرة توصيل صحيحة');
      return;
    }

    const newOrder = {
      id: Date.now().toString(),
      pickup,
      destination,
      details,
      fee: Number(fee),
      status: 'pending'
    };

    setOrders(prev => [newOrder, ...prev]);

    setPickup('');
    setDestination('');
    setDetails('');
    setFee('');

    Alert.alert('نجاح', 'تم إنشاء طلب التوصيل');
  };

  const acceptOrder = (id) => {
    setOrders(prev =>
      prev.map(order =>
        order.id === id
          ? { ...order, status: 'accepted' }
          : order
      )
    );

    Alert.alert('نجاح', 'تم قبول الطلب');
  };

  const completeOrder = (id) => {
    setOrders(prev =>
      prev.map(order =>
        order.id === id
          ? { ...order, status: 'completed' }
          : order
      )
    );

    Alert.alert('نجاح', 'تم إنهاء الطلب');
  };

  const statusText = (status) => {
    if (status === 'pending') return 'بانتظار مندوب';
    if (status === 'accepted') return 'تم قبول الطلب';
    return 'تم إكمال التوصيل';
  };

  return (
    <View style={styles.container}>
      <StatusBar backgroundColor="#123C32" barStyle="light-content" />

      <View style={styles.header}>
        <Text style={styles.logo}>وصّلني 🇮🇶</Text>
        <Text style={styles.subtitle}>خدمة التوصيل في العراق</Text>
      </View>

      <View style={styles.switch}>
        <TouchableOpacity
          style={[styles.switchButton, role === 'customer' && styles.active]}
          onPress={() => setRole('customer')}
        >
          <Text style={styles.switchText}>الزبون</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={[styles.switchButton, role === 'driver' && styles.active]}
          onPress={() => setRole('driver')}
        >
          <Text style={styles.switchText}>المندوب</Text>
        </TouchableOpacity>
      </View>

      <ScrollView contentContainerStyle={styles.content}>

        <Text style={styles.location}>📍 صلاح الدين - قضاء بيجي</Text>

        {role === 'customer' ? (
          <View>
            <Text style={styles.title}>اطلب مندوباً الآن</Text>

            <Text style={styles.label}>مكان الاستلام</Text>
            <TextInput
              style={styles.input}
              placeholder="أدخل عنوان الاستلام"
              value={pickup}
              onChangeText={setPickup}
            />

            <Text style={styles.label}>مكان التسليم</Text>
            <TextInput
              style={styles.input}
              placeholder="أدخل عنوان التسليم"
              value={destination}
              onChangeText={setDestination}
            />

            <Text style={styles.label}>تفاصيل الطلب</Text>
            <TextInput
              style={styles.input}
              placeholder="اكتب تفاصيل إضافية"
              value={details}
              onChangeText={setDetails}
            />

            <Text style={styles.label}>أجرة التوصيل بالدينار</Text>
            <TextInput
              style={styles.input}
              placeholder="مثلاً 3000"
              keyboardType="numeric"
              value={fee}
              onChangeText={setFee}
            />

            <TouchableOpacity
              style={styles.mainButton}
              onPress={createOrder}
            >
              <Text style={styles.buttonText}>إنشاء طلب التوصيل</Text>
            </TouchableOpacity>
          </View>
        ) : (
          <View>
            <Text style={styles.title}>طلبات التوصيل المتاحة</Text>

            {orders.filter(o => o.status === 'pending').length === 0 && (
              <Text style={styles.empty}>لا توجد طلبات متاحة حالياً</Text>
            )}
          </View>
        )}

        <Text style={styles.sectionTitle}>طلبات التوصيل</Text>

        {orders.length === 0 && (
          <Text style={styles.empty}>لم يتم إنشاء أي طلب بعد</Text>
        )}

        {orders.map(order => (
          <View key={order.id} style={styles.card}>
            <Text style={styles.orderTitle}>طلب توصيل #{order.id.slice(-5)}</Text>

            <Text style={styles.info}>📦 الاستلام: {order.pickup}</Text>
            <Text style={styles.info}>📍 التسليم: {order.destination}</Text>
            <Text style={styles.info}>📝 التفاصيل: {order.details || 'لا توجد'}</Text>
            <Text style={styles.price}>💰 {order.fee.toLocaleString('en-US')} د.ع</Text>
            <Text style={styles.status}>{statusText(order.status)}</Text>

            {role === 'driver' && order.status === 'pending' && (
              <TouchableOpacity
                style={styles.mainButton}
                onPress={() => acceptOrder(order.id)}
              >
                <Text style={styles.buttonText}>قبول الطلب</Text>
              </TouchableOpacity>
            )}

            {role === 'driver' && order.status === 'accepted' && (
              <TouchableOpacity
                style={styles.completeButton}
                onPress={() => completeOrder(order.id)}
              >
                <Text style={styles.buttonText}>إنهاء التوصيل</Text>
              </TouchableOpacity>
            )}
          </View>
        ))}

        <Text style={styles.footer}>
          وصّلني العراق - جميع الحقوق محفوظة
        </Text>

      </ScrollView>
    </View>
  );
}
