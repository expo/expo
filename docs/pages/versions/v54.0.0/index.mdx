import React, { useState } from 'react';
import { StyleSheet, Text, View, TextInput, TouchableOpacity, FlatList, SafeAreaView } from 'react-native';

// Weka Gemini API Key yako hapa
const GEMINI_API_KEY = "WEKA_API_KEY_YAKO"; 

interface Message {
  id: string;
  text: string;
  sender: 'user' | 'bot';
}

export default function App() {
  const [input, setInput] = useState('');
  const [messages, setMessages] = useState<Message[]>([
    { id: '1', text: 'Habari! Mimi ni BIZAI. Nimeundwa na Jaja Boy.', sender: 'bot' }
  ]);
  const [loading, setLoading] = useState(false);

  const sendMessage = async () => {
    if (!input.trim() || loading) return;
    const userMsg = input.trim();
    setInput('');
    setMessages(prev => [...prev, { id: Date.now().toString(), text: userMsg, sender: 'user' }]);
    setLoading(true);

    try {
      const response = await fetch(
        `https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${GEMINI_API_KEY}`,
        {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            contents: [{
              parts: [{
                text: `Wewe ni BIZAI, msaidizi wa biashara aliyeundwa na Jaja Boy. Jibu kila mara kwa Kiswahili ukitumia emoji. Mtumiaji akitaja kiwango cha mtaji (mfano: 'Nina 100000'), mpe mawazo ya biashara 3 yanayofaa mtaji huo.\n\nUjumbe wa mtumiaji: ${userMsg}`
              }]
            }]
          })
        }
      );
      const data = await response.json();
      const botText = data.candidates[0].content.parts[0].text;
      setMessages(prev => [...prev, { id: (Date.now() + 1).toString(), text: botText, sender: 'bot' }]);
    } catch (e) {
      setMessages(prev => [...prev, { id: (Date.now() + 1).toString(), text: 'Kuna tatizo la mtandao au API key. Tafadhali hakikisha umeweka API Key sahihi.', sender: 'bot' }]);
    } finally {
      setLoading(false);
    }
  };

  return (
    <SafeAreaView style={styles.container}>
      <View style={styles.header}><Text style={styles.headerText}>BIZAI</Text></View>
      <FlatList
        data={messages}
        keyExtractor={item => item.id}
        renderItem={({ item }) => (
          <View style={[styles.bubble, item.sender === 'user' ? styles.userBubble : styles.botBubble]}>
            <Text style={styles.bubbleText}>{item.text}</Text>
          </View>
        )}
        contentContainerStyle={styles.list}
      />
      <View style={styles.inputContainer}>
        <TextInput
          style={styles.textInput}
          placeholder="Andika hapa (mf. Nina 100000)..."
          value={input}
          onChangeText={setInput}
        />
        <TouchableOpacity style={styles.sendButton} onPress={sendMessage} disabled={loading}>
          <Text style={styles.sendText}>{loading ? '...' : 'Tuma'}</Text>
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#f5f5f5' },
  header: { backgroundColor: '#1EBE5B', padding: 15, alignItems: 'center' },
  headerText: { color: '#fff', fontSize: 20, fontWeight: 'bold' },
  list: { padding: 15 },
  bubble: { padding: 12, borderRadius: 15, marginVertical: 5, maxWidth: '80%' },
  userBubble: { alignSelf: 'flex-end', backgroundColor: '#e2fcd3' },
  botBubble: { alignSelf: 'flex-start', backgroundColor: '#fff', borderWidth: 1, borderColor: '#eee' },
  bubbleText: { fontSize: 16 },
  inputContainer: { flexDirection: 'row', padding: 10, backgroundColor: '#fff', alignItems: 'center' },
  textInput: { flex: 1, borderWidth: 1, borderColor: '#ccc', borderRadius: 20, paddingHorizontal: 15, height: 40, marginRight: 10 },
  sendButton: { backgroundColor: '#1EBE5B', borderRadius: 20, paddingHorizontal: 20, height: 40, justifyContent: 'center' },
  sendText: { color: '#fff', fontWeight: 'bold' }
});
