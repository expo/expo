Expo UI CompositionLocalProvider matches the official Jetpack Compose [`CompositionLocalProvider`](<https://developer.android.com/reference/kotlin/androidx/compose/runtime/package-summary#CompositionLocalProvider(kotlin.Array,kotlin.Function0)>) API. It provides values to all components inside it. It adds no layout of its own, so children keep modifiers such as `weight` from the parent `Row` or `Column`.

## Usage

### Smaller touch targets

Material components such as `Button`, `IconButton`, `Checkbox`, `RadioButton`, and `Switch` reserve at least 48dp of space for touch, even when they are drawn smaller. Set `minimumInteractiveComponentSize` to `0` to remove this minimum. Make sure that the controls are still easy to tap.

<ComponentExample
  title="SmallTouchTargetsExample.tsx"
  src="/static/images/expo-ui/examples/compositionlocalprovider-small-touch-targets-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/compositionlocalprovider-small-touch-targets-android-dark.webp"
  alt="A plus icon button and an empty checkbox next to each other in a row">

```tsx
import {
  Checkbox,
  CompositionLocalProvider,
  Host,
  IconButton,
  Row,
  Text,
} from '@expo/ui/jetpack-compose';
import { size } from '@expo/ui/jetpack-compose/modifiers';
import { useState } from 'react';

export default function SmallTouchTargetsExample() {
  const [checked, setChecked] = useState(false);

  return (
    <Host matchContents>
      <CompositionLocalProvider
        values={{ minimumInteractiveComponentSize: 0 }}>
        <Row
          horizontalArrangement={{ spacedBy: 8 }}
          verticalAlignment="center">
          <IconButton
            modifiers={[size(24, 24)]}
            onClick={() => console.log('Pressed')}>
            <Text>+</Text>
          </IconButton>
          <Checkbox value={checked} onCheckedChange={setChecked} />
        </Row>
      </CompositionLocalProvider>
    </Host>
  );
}
```

</ComponentExample>

### Autofill highlight

When autofill fills a text field, Compose paints a semi-transparent yellow over it. Set `autofillHighlightColor` to change that color, or to `transparent` to remove it.

```tsx AutofillHighlightExample.tsx
import {
  BasicTextField,
  CompositionLocalProvider,
  Host,
  useNativeState,
} from '@expo/ui/jetpack-compose';
import {
  background,
  fillMaxWidth,
  padding,
  semantics,
} from '@expo/ui/jetpack-compose/modifiers';

export default function AutofillHighlightExample() {
  const email = useNativeState('');

  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <CompositionLocalProvider
        values={{ autofillHighlightColor: 'transparent' }}>
        <BasicTextField
          value={email}
          singleLine
          modifiers={[
            fillMaxWidth(),
            background('#f3f4f6'),
            padding(12, 10, 12, 10),
            semantics({ contentType: 'username' }),
          ]}
        />
      </CompositionLocalProvider>
    </Host>
  );
}
```
