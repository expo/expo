Expo UI SplitButtonLayout matches the official Jetpack Compose [`SplitButtonLayout`](https://kotlinlang.org/api/compose-multiplatform/material3/androidx.compose.material3/-split-button-layout.html) API. It places a leading button for the primary action next to a trailing button, which usually opens a menu of related actions. Both buttons take the same height.

## Usage

### Split button with a menu

`SplitButtonLayout` has two slots: `SplitButtonLayout.LeadingButton` and `SplitButtonLayout.TrailingButton`. Put a `SplitButtonLeadingButton` in the leading slot. To open a menu from the trailing button, put a [`DropdownMenu`](./dropdownmenu) in the trailing slot with a `SplitButtonTrailingButton` as the menu's trigger. The menu then opens below the button. Set `checked` on the trailing button while the menu is expanded. Material 3 changes the shape of a checked button to show that the menu is open.

```tsx SplitButtonWithMenu.tsx
import {
  DropdownMenu,
  DropdownMenuItem,
  Host,
  Icon,
  SplitButtonLayout,
  SplitButtonLeadingButton,
  SplitButtonTrailingButton,
  Text,
} from '@expo/ui/jetpack-compose';
import { rotate } from '@expo/ui/jetpack-compose/modifiers';
import { useState } from 'react';

const DROP_DOWN_ICON = require('./assets/arrow_drop_down.xml');

export default function SplitButtonWithMenu() {
  const [expanded, setExpanded] = useState(false);

  return (
    <Host matchContents>
      <SplitButtonLayout>
        <SplitButtonLayout.LeadingButton>
          <SplitButtonLeadingButton
            onClick={() => console.log('Send')}>
            <Text>Send</Text>
          </SplitButtonLeadingButton>
        </SplitButtonLayout.LeadingButton>
        <SplitButtonLayout.TrailingButton>
          <DropdownMenu
            expanded={expanded}
            onDismissRequest={() => setExpanded(false)}>
            <DropdownMenu.Trigger>
              <SplitButtonTrailingButton
                checked={expanded}
                onCheckedChange={setExpanded}>
                <Icon
                  source={DROP_DOWN_ICON}
                  contentDescription="Send options"
                  modifiers={[rotate(expanded ? 180 : 0)]}
                />
              </SplitButtonTrailingButton>
            </DropdownMenu.Trigger>
            <DropdownMenu.Items>
              <DropdownMenuItem onClick={() => setExpanded(false)}>
                <DropdownMenuItem.Text>
                  <Text>Schedule send</Text>
                </DropdownMenuItem.Text>
              </DropdownMenuItem>
            </DropdownMenu.Items>
          </DropdownMenu>
        </SplitButtonLayout.TrailingButton>
      </SplitButtonLayout>
    </Host>
  );
}
```

The slots accept other content too. For example, wrap the trailing button in a [`TooltipBox`](./tooltip) to show a tooltip on long press.

### Full-width split button

The layout measures the trailing button first and gives the leading button the remaining width. Apply `fillMaxWidth()` to both the layout and the leading button to make the split button span its container.

```tsx FullWidthSplitButton.tsx
import {
  Host,
  Icon,
  SplitButtonLayout,
  SplitButtonLeadingButton,
  SplitButtonTrailingButton,
  Text,
} from '@expo/ui/jetpack-compose';
import { fillMaxWidth } from '@expo/ui/jetpack-compose/modifiers';

const DROP_DOWN_ICON = require('./assets/arrow_drop_down.xml');

export default function FullWidthSplitButton() {
  return (
    <Host
      matchContents={{ vertical: true }}
      style={{ width: '100%' }}>
      <SplitButtonLayout modifiers={[fillMaxWidth()]}>
        <SplitButtonLayout.LeadingButton>
          <SplitButtonLeadingButton
            onClick={() => console.log('Merge')}
            modifiers={[fillMaxWidth()]}>
            <Text>Squash and merge</Text>
          </SplitButtonLeadingButton>
        </SplitButtonLayout.LeadingButton>
        <SplitButtonLayout.TrailingButton>
          <SplitButtonTrailingButton checked={false}>
            <Icon
              source={DROP_DOWN_ICON}
              contentDescription="Merge options"
            />
          </SplitButtonTrailingButton>
        </SplitButtonLayout.TrailingButton>
      </SplitButtonLayout>
    </Host>
  );
}
```
