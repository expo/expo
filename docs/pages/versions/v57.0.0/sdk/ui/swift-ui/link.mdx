## Usage

### Basic link

<ComponentExample
  title="BasicLinkExample.tsx"
  src="/static/images/expo-ui/examples/link-basic-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/link-basic-ios-dark.webp"
  alt="A blue Visit Expo text link">

```tsx
import { Host, Link } from '@expo/ui/swift-ui';

export default function BasicLinkExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Link label="Visit Expo" destination="https://expo.dev" />
    </Host>
  );
}
```

</ComponentExample>

### Custom label content

You can pass custom components as `children` for more complex link label content.

<ComponentExample
  title="CustomContentExample.tsx"
  src="/static/images/expo-ui/examples/link-custom-content-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/link-custom-content-ios-dark.webp"
  alt="A link SF Symbol above the text Expo, both tinted blue">

```tsx
import { Host, Link, VStack, Image, Text } from '@expo/ui/swift-ui';

export default function CustomContentExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Link destination="https://expo.dev">
        <VStack spacing={4}>
          <Image systemName="link" />
          <Text>Expo</Text>
        </VStack>
      </Link>
    </Host>
  );
}
```

</ComponentExample>
