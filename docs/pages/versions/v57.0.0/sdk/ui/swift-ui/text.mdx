### Text with modifiers

Use modifiers to style the entire text.

<ComponentExample
  title="StyledTextExample.tsx"
  src="/static/images/expo-ui/examples/text-styled-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-styled-ios-dark.webp"
  alt="The words Large Bold Blue Text rendered in a large bold blue font">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import { font, foregroundStyle } from '@expo/ui/swift-ui/modifiers';

export default function StyledTextExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text
        modifiers={[
          font({ size: 24, weight: 'bold' }),
          foregroundStyle('blue'),
        ]}>
        Large Bold Blue Text
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### Nested text (per-segment styling)

Nest `Text` components to style individual segments differently. This is useful for inline formatting, such as bold or colored words within a sentence.

> **Note:** Nested text uses SwiftUI's [Text concatenation](https://developer.apple.com/documentation/swiftui/text), so only modifiers that return `Text` (such as `bold`, `italic`, `font`, `foregroundColor`, and `foregroundStyle` with color) will apply to nested segments.

<ComponentExample
  title="NestedTextExample.tsx"
  src="/static/images/expo-ui/examples/text-nested-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-nested-ios-dark.webp"
  alt="The sentence Hello world! where the word world is bold and red">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import {
  bold,
  italic,
  foregroundStyle,
} from '@expo/ui/swift-ui/modifiers';

export default function NestedTextExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text>
        Hello{' '}
        <Text modifiers={[bold(), foregroundStyle('red')]}>
          world
        </Text>
        !
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### Mixed inline styles

Combine multiple styled segments for rich text formatting.

<ComponentExample
  title="MixedStylesExample.tsx"
  src="/static/images/expo-ui/examples/text-mixed-styles-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-mixed-styles-ios-dark.webp"
  alt="A sentence where the word bold is bold, italic is italic, and colored is orange">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import {
  bold,
  italic,
  foregroundStyle,
  font,
} from '@expo/ui/swift-ui/modifiers';

export default function MixedStylesExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text>
        This is <Text modifiers={[bold()]}>bold</Text>,{' '}
        <Text modifiers={[italic()]}>italic</Text>, and{' '}
        <Text modifiers={[foregroundStyle('orange')]}>colored</Text>{' '}
        text.
      </Text>
    </Host>
  );
}
```

</ComponentExample>

### Font weights

Use the `font` modifier to apply different font weights.

<ComponentExample
  title="FontWeightsExample.tsx"

```tsx
          Ultra Light

        <Text modifiers={[font({ weight: 'regular' })]}>

        <Text modifiers={[font({ weight: 'black' })]}>Black</Text>
      </VStack>
```


</ComponentExample>

### Custom fonts

Use the `font` modifier with a `family` parameter to use custom fonts. You can load custom fonts using [`expo-font`](/versions/latest/sdk/font/) library.

```tsx CustomFontExample.tsx
import { Host, Text, VStack } from '@expo/ui/swift-ui';
import { font } from '@expo/ui/swift-ui/modifiers';

export default function CustomFontExample() {
  return (
    <Host matchContents>
      <VStack spacing={4}>
        <Text
          modifiers={[font({ family: 'Inter-Bold', size: 18 })]}>
          Inter Bold
        </Text>
        <Text
          modifiers={[font({ family: 'Inter-Regular', size: 18 })]}>
          Inter Regular
        </Text>
      </VStack>
    </Host>
  );
}
```

### Text with line limit

Use the `lineLimit` modifier to truncate text after a certain number of lines.

<ComponentExample
  title="LineLimitExample.tsx"
  src="/static/images/expo-ui/examples/text-line-limit-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-line-limit-ios-dark.webp"
  alt="A long paragraph cut off after two lines with a trailing ellipsis">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';
import { lineLimit } from '@expo/ui/swift-ui/modifiers';

export default function LineLimitExample() {
  const longText =
    'This is a very long text that will be truncated after two lines. '.repeat(
      5
    );

  return (
    <Host style={{ flex: 1 }}>
      <Text modifiers={[lineLimit(2)]}>{longText}</Text>
    </Host>
  );
}
```

</ComponentExample>

### Auto-updating date

Use the `date` and `dateStyle` props to display a date that automatically updates as time passes. This is especially useful in widgets and Live Activities.

<ComponentExample
  title="DateTextExample.tsx"
  src="/static/images/expo-ui/examples/text-date-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-date-ios-dark.webp"
  alt="A countdown reading 4 minutes and 53 seconds">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';

export default function DateTextExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text
        date={new Date(Date.now() + 300000)}
        dateStyle="timer"
      />
    </Host>
  );
}
```

</ComponentExample>

### Timer interval

Use `timerInterval` to display a live countdown or count-up timer. This requires iOS/tvOS 16+.

<ComponentExample
  title="TimerIntervalExample.tsx"
  src="/static/images/expo-ui/examples/text-timer-interval-ios-light.webp"
  darkSrc="/static/images/expo-ui/examples/text-timer-interval-ios-dark.webp"
  alt="A countdown reading 9 minutes and 54 seconds">

```tsx
import { Host, Text } from '@expo/ui/swift-ui';

export default function TimerIntervalExample() {
  return (
    <Host style={{ flex: 1 }}>
      <Text
        timerInterval={{
          lower: new Date(),
          upper: new Date(Date.now() + 600000),
        }}
        countsDown
      />
    </Host>
  );
}
```

</ComponentExample>

> **Note:** `timerInterval`, `countsDown`, and `pauseTime` require iOS 16.0+ / tvOS 16.0+. On older versions, the timer interval will not render.
