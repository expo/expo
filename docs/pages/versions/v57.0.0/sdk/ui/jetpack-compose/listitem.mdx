## Usage

### Basic list item

<ComponentExample
  title="BasicListItem.tsx"
  src="/static/images/expo-ui/examples/listitem-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/listitem-basic-android-dark.webp"
  alt="A single list row with the headline Settings">

```tsx
import { Host, ListItem, Text } from '@expo/ui/jetpack-compose';

export default function BasicListItem() {
  return (
    <Host matchContents>
      <ListItem>
        <ListItem.HeadlineContent>
          <Text>Settings</Text>
        </ListItem.HeadlineContent>
      </ListItem>
    </Host>
  );
}
```

</ComponentExample>

### With compound components

Use compound components for rich content in each position.

<ComponentExample
  title="ListItemWithCompoundComponent.tsx"
  src="/static/images/expo-ui/examples/listitem-slots-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/listitem-slots-android-dark.webp"
  alt="A list row with a bell icon, an ACCOUNT overline, a Notifications headline, supporting text, and a trailing chevron">

```tsx
import {
  Host,
  ListItem,
  Icon,
  Text,
} from '@expo/ui/jetpack-compose';

export default function ListItemWithSlots() {
  return (
    <Host matchContents>
      <ListItem>
        <ListItem.HeadlineContent>
          <Text>Notifications</Text>
        </ListItem.HeadlineContent>
        <ListItem.OverlineContent>
          <Text>ACCOUNT</Text>
        </ListItem.OverlineContent>
        <ListItem.SupportingContent>
          <Text>Manage notification preferences</Text>
        </ListItem.SupportingContent>
        <ListItem.LeadingContent>
          <Icon source={require('./assets/notifications.xml')} />
        </ListItem.LeadingContent>
        <ListItem.TrailingContent>
          <Icon source={require('./assets/chevron.xml')} />
        </ListItem.TrailingContent>
      </ListItem>
    </Host>
  );
}
```

</ComponentExample>

### Clickable list item

Use the `clickable` modifier to handle tap interactions.

<ComponentExample
  title="ClickableListItem.tsx"
  src="/static/images/expo-ui/examples/listitem-clickable-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/listitem-clickable-android-dark.webp"
  alt="A tappable list row with the headline Tap me">

```tsx
import { Host, ListItem, Text } from '@expo/ui/jetpack-compose';
import { clickable } from '@expo/ui/jetpack-compose/modifiers';

export default function ClickableListItem() {
  return (
    <Host matchContents>
      <ListItem
        modifiers={[clickable(() => console.log('Tapped!'))]}>
        <ListItem.HeadlineContent>
          <Text>Tap me</Text>
        </ListItem.HeadlineContent>
      </ListItem>
    </Host>
  );
}
```

</ComponentExample>

### Custom headline content

Use `ListItem.HeadlineContent` for composable headline content like rows with icons.

<ComponentExample
  title="ListItemCustomHeadline.tsx"
  src="/static/images/expo-ui/examples/listitem-custom-headline-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/listitem-custom-headline-android-dark.webp"
  alt="A list row whose headline pairs the text Premium Feature with a star icon">

```tsx
import {
  Host,
  ListItem,
  Text,
  Row,
  Icon,
} from '@expo/ui/jetpack-compose';

export default function ListItemCustomHeadline() {
  return (
    <Host matchContents>
      <ListItem>
        <ListItem.HeadlineContent>
          <Row
            horizontalArrangement={{ spacedBy: 8 }}
            verticalAlignment="center">
            <Text>Premium Feature</Text>
            <Icon source={require('./assets/star.xml')} size={16} />
          </Row>
        </ListItem.HeadlineContent>
      </ListItem>
    </Host>
  );
}
```

</ComponentExample>
