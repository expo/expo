### Variants

Pick a visual variant with the [`variant`](#variant) prop.

<ComponentExample
  title="ButtonVariantsExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-button-variants-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-button-variants-android-dark.webp',
    alt: 'Filled, outlined, and text button variants stacked vertically',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-button-variants-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-button-variants-ios-dark.webp',
    alt: 'Filled, outlined, and text button variants stacked vertically',
  }}>

```tsx
import { Host, Column, Button } from '@expo/ui';

export default function ButtonVariantsExample() {
  return (
    <Host matchContents>
      <Column spacing={8}>
        <Button variant="filled" label="Filled" onPress={() => {}} />
        <Button variant="outlined" label="Outlined" onPress={() => {}} />
        <Button variant="text" label="Text" onPress={() => {}} />
      </Column>
    </Host>
  );
}
```

</ComponentExample>

### Custom content

Pass [`children`](#children) for fully custom button contents. The [`label`](#label) prop is ignored when `children` is provided.

<ComponentExample
  title="CustomButtonExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-button-custom-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-button-custom-android-dark.webp',
    alt: 'A filled button with a star icon and a Favorite label',
  }}
  ios={{
    src: '/static/images/expo-ui/examples/universal-button-custom-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-button-custom-ios-dark.webp',
    alt: 'A filled button with a star icon and a Favorite label',
  }}>

```tsx
import { Host, Button, Row, Icon, Text } from '@expo/ui';

export default function CustomButtonExample() {
  return (
    <Host matchContents>
      <Button onPress={() => {}}>
        <Row spacing={6} alignment="center">
          <Icon
            name={Icon.select({
              ios: 'star.fill',
              android: require('@expo/material-symbols/star.xml'),
            })}
            size={16}
            color="#FFFFFF"
          />
          <Text textStyle={{ color: '#FFFFFF' }}>Favorite</Text>
        </Row>
      </Button>
    </Host>
  );
}
```

</ComponentExample>
