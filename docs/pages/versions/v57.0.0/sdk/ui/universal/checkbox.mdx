### Basic checkbox

<ComponentExample
  title="CheckboxExample.tsx"
  android={{
    src: '/static/images/expo-ui/examples/universal-checkbox-basic-android-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-checkbox-basic-android-dark.webp',
    alt: 'An unchecked checkbox with a consent label',

  ios={{
    src: '/static/images/expo-ui/examples/universal-checkbox-basic-ios-light.webp',
    darkSrc: '/static/images/expo-ui/examples/universal-checkbox-basic-ios-dark.webp',
    alt: 'A switch-style checkbox with a consent label',
  }}>

```tsx
import { useState } from 'react';
import { Host, Checkbox } from '@expo/ui';

export default function CheckboxExample() {
  const [accepted, setAccepted] = useState(false);

  return (
    <Host matchContents>
      <Checkbox label="I accept the terms" value={accepted} onValueChange={setAccepted} />
    </Host>
  );
}
```

</ComponentExample>
