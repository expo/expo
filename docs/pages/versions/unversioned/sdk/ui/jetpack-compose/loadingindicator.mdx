## Usage

### Loading indicator

A morphing-shape loading animation from Material 3 Expressive.

<ComponentExample
  title="LoadingIndicatorExample.tsx"
  src="/static/images/expo-ui/examples/loadingindicator-basic-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/loadingindicator-basic-android-dark.webp"
  alt="A single morphing shape loading indicator">

```tsx
import { Host, LoadingIndicator } from '@expo/ui/jetpack-compose';

export default function LoadingIndicatorExample() {
  return (
    <Host matchContents>
      <LoadingIndicator />
    </Host>
  );
}
```

</ComponentExample>

### Indeterminate

Omit the `progress` prop to animate continuously without indicating a specific completion level.

<ComponentExample
  title="IndeterminateExample.tsx"
  src="/static/images/expo-ui/examples/loadingindicator-indeterminate-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/loadingindicator-indeterminate-android-dark.webp"
  alt="A plain loading indicator beside a contained one, both animating continuously">

```tsx
import {
  ContainedLoadingIndicator,
  Host,
  LoadingIndicator,
  Row,
} from '@expo/ui/jetpack-compose';

export default function IndeterminateExample() {
  return (
    <Host matchContents>
      <Row horizontalArrangement={{ spacedBy: 16 }}>
        <LoadingIndicator />
        <ContainedLoadingIndicator />
      </Row>
    </Host>
  );
}
```

</ComponentExample>

### Determinate

Pass an observable state from `useNativeState` as `progress`. Update `progress.value` between `0` and `1`.

<ComponentExample
  title="DeterminateExample.tsx"
  src="/static/images/expo-ui/examples/loadingindicator-determinate-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/loadingindicator-determinate-android-dark.webp"
  alt="A plain and a contained loading indicator, each drawing the shape for the current progress value">

```tsx
import {
  ContainedLoadingIndicator,
  Host,
  LoadingIndicator,
  Row,
  useNativeState,
} from '@expo/ui/jetpack-compose';
import { useEffect } from 'react';

export default function DeterminateExample() {
  const progress = useNativeState(0);

  useEffect(() => {
    const interval = setInterval(() => {
      progress.value = (progress.value + 0.05) % 1;
    }, 500);
    return () => clearInterval(interval);
  }, [progress]);

  return (
    <Host matchContents>
      <Row horizontalArrangement={{ spacedBy: 16 }}>
        <LoadingIndicator progress={progress} />
        <ContainedLoadingIndicator progress={progress} />
      </Row>
    </Host>
  );
}
```

</ComponentExample>
