## Usage

### Linear progress

A horizontal bar that fills to indicate progress. Provide a `progress` value between `0` and `1` for determinate mode.

<ComponentExample
  title="LinearExample.tsx"
  src="/static/images/expo-ui/examples/progress-linear-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/progress-linear-android-dark.webp"
  alt="A horizontal progress bar filled to half its track">

```tsx
import {
  Host,
  LinearProgressIndicator,
} from '@expo/ui/jetpack-compose';

export default function LinearExample() {
  return (
    <Host matchContents>
      <LinearProgressIndicator progress={0.5} />
    </Host>
  );
}
```

</ComponentExample>

### Circular progress

A spinning circle whose stroke grows to indicate progress.

<ComponentExample
  title="CircularExample.tsx"
  src="/static/images/expo-ui/examples/progress-circular-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/progress-circular-android-dark.webp"
  alt="A circular progress ring drawn three quarters of the way round">

```tsx
import {
  Host,
  CircularProgressIndicator,
} from '@expo/ui/jetpack-compose';

export default function CircularExample() {
  return (
    <Host matchContents>
      <CircularProgressIndicator progress={0.75} />
    </Host>
  );
}
```

</ComponentExample>

### Indeterminate

Omit the `progress` prop to animate continuously without indicating a specific completion level.

<ComponentExample
  title="IndeterminateExample.tsx"
  src="/static/images/expo-ui/examples/progress-indeterminate-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/progress-indeterminate-android-dark.webp"
  alt="Linear, circular, wavy circular, and wavy linear indicators animating without a set value">

```tsx
import {
  CircularProgressIndicator,
  CircularWavyProgressIndicator,
  Column,
  Host,
  LinearProgressIndicator,
  LinearWavyProgressIndicator,
} from '@expo/ui/jetpack-compose';

export default function IndeterminateExample() {
  return (
    <Host matchContents>
      <Column verticalArrangement={{ spacedBy: 16 }}>
        <LinearProgressIndicator />
        <CircularProgressIndicator />
        <CircularWavyProgressIndicator />
        <LinearWavyProgressIndicator />
      </Column>
    </Host>
  );
}
```

</ComponentExample>

### Custom colors

Use `color` for the indicator and `trackColor` for the background track.

<ComponentExample
  title="ColorsExample.tsx"
  src="/static/images/expo-ui/examples/progress-colors-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/progress-colors-android-dark.webp"
  alt="A circular progress ring drawn in red over a grey track">

```tsx
import {
  Host,
  CircularProgressIndicator,
} from '@expo/ui/jetpack-compose';

export default function ColorsExample() {
  return (
    <Host matchContents>
      <CircularProgressIndicator
        progress={0.6}
        color="red"
        trackColor="#cccccc"
      />
    </Host>
  );
}
```

</ComponentExample>

### Wavy variants

`LinearWavyProgressIndicator` and `CircularWavyProgressIndicator` add an expressive wave animation from Material 3 Expressive.

<ComponentExample
  title="WavyExample.tsx"
  src="/static/images/expo-ui/examples/progress-wavy-android-light.webp"
  darkSrc="/static/images/expo-ui/examples/progress-wavy-android-dark.webp"
  alt="A wavy linear bar and a wavy circular ring, each at sixty percent">

```tsx
import {
  Host,
  LinearWavyProgressIndicator,
  CircularWavyProgressIndicator,
  Column,
} from '@expo/ui/jetpack-compose';

export default function WavyExample() {
  return (
    <Host matchContents>
      <Column verticalArrangement={{ spacedBy: 16 }}>
        <LinearWavyProgressIndicator progress={0.6} />
        <CircularWavyProgressIndicator progress={0.6} />
      </Column>
    </Host>
  );
}
```

</ComponentExample>
