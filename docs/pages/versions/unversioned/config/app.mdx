
import React, { useState } from 'react';
import { View, Text, StyleSheet, TouchableOpacity } from 'react-native';
import { CameraView, useCameraPermissions } from 'expo-camera';

export default function App() {
  const [permission, requestPermission] = useCameraPermissions();
  const [flash, setFlash] = useState('auto');
  const [filter, setFilter] = useState('Natural');

  if (!permission) {
    return <View style={styles.center}><Text>Cargando...</Text></View>;
  }

  if (!permission.granted) {
    return (
      <View style={styles.center}>
        <Text style={styles.text}>Necesitamos permiso para usar la cámara</Text>
        <TouchableOpacity style={styles.button} onPress={requestPermission}>
          <Text style={styles.text}>Permitir cámara</Text>
        </TouchableOpacity>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <CameraView style={styles.camera} facing="back" flash={flash}>
        <View style={styles.top}>
          <Text style={styles.title}>MI CÁMARA</Text>
          <TouchableOpacity
            style={styles.button}
            onPress={() => setFlash(
              flash === 'auto' ? 'on' : flash === 'on' ? 'off' : 'auto'
            )}
          >
            <Text style={styles.text}>Flash: {flash.toUpperCase()}</Text>
          </TouchableOpacity>
        </View>

        <View style={styles.bottom}>
          <Text style={styles.text}>Filtro: {filter}</Text>
          <View style={styles.filters}>
            {['Natural', 'Vivo', 'Cálido', 'Frío', 'B/N'].map((item) => (
              <TouchableOpacity
                key={item}
                style={[styles.button, filter === item && styles.selected]}
                onPress={() => setFilter(item)}
              >
                <Text style={styles.text}>{item}</Text>
              </TouchableOpacity>
            ))}
          </View>
          <Text style={styles.hint}>Toca el botón para tomar una foto</Text>
        </View>
      </CameraView>
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, backgroundColor: '#000' },
  camera: { flex: 1, justifyContent: 'space-between' },
  center: {
    flex: 1, backgroundColor: '#111',
    alignItems: 'center', justifyContent: 'center', padding: 20
  },
  top: {
    paddingTop: 45, paddingHorizontal: 18,
    flexDirection: 'row', justifyContent: 'space-between',
    alignItems: 'center'
  },
  bottom: { padding: 18, alignItems: 'center', gap: 12 },
  title: { color: 'white', fontSize: 20, fontWeight: 'bold' },
  text: { color: 'white', fontSize: 14 },
  button: {
    backgroundColor: '#333', padding: 12,
    borderRadius: 20, margin: 3
  },
  selected: { backgroundColor: '#1677ff' },
  filters: {
    flexDirection: 'row', flexWrap: 'wrap', justifyContent: 'center'
  },
  hint: { color: 'white', marginBottom: 12 }
});
