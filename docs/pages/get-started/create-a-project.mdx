/* ---------- انیمیشن‌های صفحه‌های اصلی ---------- */

function Bubbles({ count = 14, color = 'rgba(255,255,255,0.16)' }) {
  const items = useMemo(
    () =>
      Array.from({ length: count }, () => ({
        left: Math.random() * W,
        size: 8 + Math.random() * 30,
        dur: 6000 + Math.random() * 7000,
        delay: Math.random() * 4000,
        v: new Animated.Value(0),
      })),
    [count]
  );
  useEffect(() => {
    const loops = items.map((it) => {
      const l = Animated.loop(
        Animated.timing(it.v, {
          toValue: 1,
          duration: it.dur,
          delay: it.delay,
          easing: Easing.linear,
          useNativeDriver: true,
        })
      );
      l.start();
      return l;
    });
    return () => loops.forEach((l) => l.stop());
  }, [items]);
  return (
    <View pointerEvents="none" style={StyleSheet.absoluteFill}>
      {items.map((it, i) => (
        <Animated.View
          key={i}
          style={{
            position: 'absolute',
            left: it.left,
            bottom: -50,
            width: it.size,
            height: it.size,
            borderRadius: it.size / 2,
            backgroundColor: color,
            opacity: it.v.interpolate({ inputRange: [0, 0.15, 0.85, 1], outputRange: [0, 1, 1, 0] }),
            transform: [{ translateY: it.v.interpolate({ inputRange: [0, 1], outputRange: [0, -(H + 100)] }) }],
          }}
        />
      ))}
    </View>
  );
}

function Floating({ children, delay = 0, amp = 12, style }) {
  const v = useRef(new Animated.Value(0)).current;
  useEffect(() => {
    const l = Animated.loop(
      Animated.sequence([
        Animated.timing(v, { toValue: 1, duration: 1800, delay, easing: Easing.inOut(Easing.sin), useNativeDriver: true }),
        Animated.timing(v, { toValue: 0, duration: 1800, easing: Easing.inOut(Easing.sin), useNativeDriver: true }),
      ])
    );
    l.start();
    return () => l.stop();
  }, [v, delay]);
  return (
    <Animated.View
      style={[
        style,
        {
          transform: [
            { translateY: v.interpolate({ inputRange: [0, 1], outputRange: [0, -amp] }) },
            { rotate: v.interpolate({ inputRange: [0, 1], outputRange: ['-5deg', '5deg'] }) },
          ],
        },
      ]}
    >
      {children}
    </Animated.View>
  );
}

function Reveal({ delay = 0, children, style, from = 26 }) {
  const v = useRef(new Animated.Value(0)).current;
  useEffect(() => {
    Animated.timing(v, {
      toValue: 1,
      duration: 650,
      delay,
      easing: Easing.out(Easing.cubic),
      useNativeDriver: true,
    }).start();
  }, [v, delay]);
  return (
    <Animated.View
      style={[
        style,
        {
          opacity: v,
          transform: [{ translateY: v.interpolate({ inputRange: [0, 1], outputRange: [from, 0] }) }],
        },
      ]}
    >
      {children}
    </Animated.View>
  );
}

function Pulse({ size, color }) {
  const v = useRef(new Animated.Value(0)).current;
  useEffect(() => {
    const l = Animated.loop(
      Animated.timing(v, { toValue: 1, duration: 2200, easing: Easing.out(Easing.quad), useNativeDriver: true })
    );
    l.start();
    return () => l.stop();
  }, [v]);
  return (
    <Animated.View
      pointerEvents="none"
      style={{
        position: 'absolute',
        width: size,
        height: size,
        borderRadius: size / 2,
        borderWidth: 2,
        borderColor: color,
        opacity: v.interpolate({ inputRange: [0, 1], outputRange: [0.7, 0] }),
        transform: [{ scale: v.interpolate({ inputRange: [0, 1], outputRange: [1, 1.9] }) }],
      }}
    />
  );
}/* ---------- صفحه شروع ---------- */

/* ---------- صفحه جدا برای هر مرحله ---------- */

function Detail({ index, onBack, onGo }) {
  const s = STAGES[index];
  const total = STAGES.length;
  const handlers = useRef({});
  handlers.current = {
    next: () => index < total - 1 && onGo(index + 1),
    prev: () => index > 0 && onGo(index - 1),
  };

  const pan = useRef(
    PanResponder.create({
      onMoveShouldSetPanResponder: (_, g) => Math.abs(g.dx) > 25 && Math.abs(g.dx) > Math.abs(g.dy) * 2,
      onPanResponderRelease: (_, g) => {
        if (g.dx < -70) handlers.current.next();
        else if (g.dx > 70) handlers.current.prev();
      },
    })
  ).current;

  return (
    <LinearGradient colors={['#0b1d2e']} style={styles.fill} {...pan.panHandlers}>
      <View style={styles.topBar}>
        <TouchableOpacity onPress={onBack} style={styles.backBtn}>
          <Text style={styles.backTxt}>فهرست مراحل ›</Text>
        </TouchableOpacity>
        <Text style={styles.counter}>
          {toFa(index + 1)} از {toFa(total)}
        </Text>
      </View>

      <View style={styles.progressTrack}>
        <View style={[styles.progressFill, { width: `${((index + 1) / total) * 100}%`, backgroundColor: s.colors[1] }]} />
      </View>

      <ScrollView contentContainerStyle={{ padding: 18, paddingBottom: 110 }} showsVerticalScrollIndicator={false}>
        {s.image ? (
          <Image source={s.image} style={styles.heroImg} resizeMode="cover" />
        ) : (
          <Scene id={s.id} color={s.colors[1]} />
        )}

        <Reveal delay={100} style={{ marginTop: 16 }}>
          <Text style={[styles.stageNum, { color: s.colors[1] }]}>مرحله {toFa(s.id)}</Text>
          <Text style={styles.stageTitle}>{s.title}</Text>
          <Text style={styles.stageShort}>{s.short}</Text>
        </Reveal>

        <Reveal delay={250}>
          <View style={styles.statsRow}>
            {s.stats.map(([k, v]) => (
              <View key={k} style={[styles.stat, { borderColor: s.colors[1] }]}>
                <Text style={styles.statK}>{k}</Text>
                <Text style={styles.statV}>{v}</Text>
              </View>
            ))}
          </View>
        </Reveal>

        <Reveal delay={400}>
          <View style={styles.textCard}>
            {s.paras.map((p, i) => (
              <Text key={i} style={[styles.stageText, i > 0 && { marginTop: 12 }]}>
                {p}
              </Text>
            ))}
          </View>
        </Reveal>

        <Reveal delay={550}>
          <Text style={styles.sectionHead}>مراحل انجام کار</Text>
          {s.steps.map((p, i) => (
            <View key={i} style={styles.pointRow}>
              <View style={[styles.numBadge, { backgroundColor: s.colors[1] }]}>
                <Text style={styles.numTxt}>{toFa(i + 1)}</Text>
              </View>
              <Text style={styles.pointTxt}>{p}</Text>
            </View>
          ))}
        </Reveal>

        <Reveal delay={650}>
          <Text style={styles.sectionHead}>تجهیزات اصلی</Text>
          <View style={styles.chips}>
            {s.equip.map((e) => (
              <View key={e} style={[styles.chip, { borderColor: s.colors[1] }]}>
                <Text style={styles.chipTxt}>{e}</Text>
              </View>
            ))}
          </View>
        </Reveal>

        <Reveal delay={750}>
          <View style={styles.factBox}>
            <Text style={styles.factHead}>💡 آیا می‌دانستید؟</Text>
            <Text style={styles.factTxt}>{s.fact}</Text>
          </View>
        </Reveal>

        <Text style={styles.swipeHint}>برای رفتن به مرحله بعد، صفحه را بکشید</Text>
      </ScrollView>

      <View style={styles.navBar}>
        <TouchableOpacity
          disabled={index === 0}
          onPress={handlers.current.prev}
          style={[styles.navBtn, { opacity: index === 0 ? 0.3 : 1 }]}

          <Text style={styles.navTxt}>› قبلی</Text>
        </TouchableOpacity>
        <TouchableOpacity
          onPress={index === total - 1 ? onBack : handlers.current.next}
          style={[styles.navBtn, { backgroundColor: s.colors[1] }]}
        >
          <Text style={[styles.navTxt, { color: '#06121f' }]}>

export default function App() {
  const [screen, setScreen] = useState('splash');
  const [index, setIndex] = useState(0);
  const screenRef = useRef(screen);
  screenRef.current = screen;

  useEffect(() => {
    const sub = BackHandler.addEventListener('hardwareBackPress', () => {
      if (screenRef.current === 'detail' || screenRef.current === 'about') {
        setScreen('home');
        return true;
      }
      return false;
    });
    return () => sub.remove();
  }, []);

  const enter = useRef(() => setScreen((p) => (p === 'splash' ? 'home' : p))).current;

  return (
    <View style={styles.fill}>
      <StatusBar />
      {screen === 'splash' && <Splash onEnter={enter} />}
      {screen === 'home' && (
        <Home
          onOpen={(i) => {
            setIndex(i);
            setScreen('detail');
          }}
          onAbout={() => setScreen('about')}
        />
      )}
      {screen === 'detail' && <Detail key={index} index={index} onBack={() => setScreen('home')} onGo={setIndex} />}
      {screen === 'about' && <About onBack={() => setScreen('home')} />}
    </View>
  );
}
