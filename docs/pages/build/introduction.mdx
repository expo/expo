export default function App() {
  const [recherche, setRecherche] = useState('');
  const [categorie, setCategorie] = useState('Tous');

  const [favoris, setFavoris] = useState([]);

    (categorie === 'Tous' || p.categorie === categorie) &&

    setFavoris((anciens) =>
      anciens.includes(id)

              value={recherche}

            <View style={styles.banner}>
              <Text style={styles.bannerTitle}>GAME PLUS+</Text>
              <Text style={styles.bannerText}>
                Consoles • Jeux • Accessoires
              </Text>

                Jouez sans limites !
              </Text>
            </View>

            <Text style={[styles.sectionTitle, { color: texte }]}>
              Nos catégories
            </Text>

            <ScrollView horizontal showsHorizontalScrollIndicator={false}>

                  style={[

            <Text style={[styles.sectionTitle, { color: texte }]}>

                    <TouchableOpacity

                      <Text style={styles.white}>Ajouter au panier</Text>
                    </TouchableOpacity>
                    <TouchableOpacity onPress={() => ajouterFavori(p.id)}>

                        {favoris.includes(p.id) ? '❤️' : '♡'}
                      </Text>
                    </TouchableOpacity>
                  </View>
                </View>
              </View>
            ))}
          </>
        )}

        {onglet === 'Catégories' && (
          <>
            <Text style={[styles.sectionTitle, { color: texte }]}>
              Toutes nos catégories
            </Text>
            {categories.slice(1).map((cat) => (
              <TouchableOpacity

                style={[styles.categoryRow, { backgroundColor: carte }]}
                onPress={() => {
                  setCategorie(cat);
                  setOnglet('Accueil');
                }}
              >
                <Text style={{ color: texte, fontWeight: 'bold' }}>{cat}</Text>
                <Text style={styles.blue}>Voir →</Text>
              </TouchableOpacity>
            ))}
          </>
        )}

        {onglet === 'Favoris' && (

            <Text style={[styles.sectionTitle, { color: texte }]}>Mes favoris</Text>

              </Text>
            ))}
            {favoris.length === 0 && (
              <Text style={{ color: texte }}>Aucun favori pour le moment.</Text>
            )}
          </>
        )}

        {onglet === 'Messagerie' && (
          <View style={[styles.product, { backgroundColor: carte }]}>
            <Text style={[styles.productName, { color: texte }]}>
              Messagerie GAME PLUS+
            </Text>
            <Text style={styles.description}>
              Bonjour et bienvenue chez GAME PLUS+ !
              Laissez-nous votre demande. Notre équipe vous répondra.
            </Text>
            <Text style={styles.blue}>WhatsApp : +224 626 00 46 90</Text>
            <Text style={styles.blue}>Téléphone : +224 669 54 88 95</Text>
          </View>
        )}

        {onglet === 'Profil' && (
          <View style={[styles.product, { backgroundColor: carte }]}>
            <Text style={[styles.productName, { color: texte }]}>
              GAME PLUS+
            </Text>
            <Text style={styles.description}>
              Madina, Centre Commercial Diané, Conakry, Guinée.
            </Text>
            <Text style={styles.description}>
              Livraison disponible en Afrique et en Europe.
            </Text>
            <TouchableOpacity
              style={styles.orderButton}
              onPress={() => setModeSombre(!modeSombre)}
            >
              <Text style={styles.white}>
                {modeSombre ? 'Activer le mode clair' : 'Activer le mode sombre'}
              </Text>
            </TouchableOpacity>
          </View>
        )}
      </ScrollView>

      <View style={styles.navigation}>
        {['Accueil', 'Catégories', 'Favoris', 'Messagerie', 'Profil'].map((item, i) => (
          <TouchableOpacity
            key={item}
            style={styles.navItem}
            onPress={() => setOnglet(item)}
          >
            <Text style={{ fontSize: 20 }}>
              {['⌂', '▦', '♡', '✉', '♙'][i]}
            </Text>
            <Text style={{
              fontSize: 10,
              color: onglet === item ? '#1769E8' : '#777',
              fontWeight: 'bold',
            }}>
              {item}
            </Text>
          </TouchableOpacity>
        ))}
      </View>
    </SafeAreaView>

}
