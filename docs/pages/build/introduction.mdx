
import React, { useState, useEffect } from "react";
import {
  View, Text, TextInput, ScrollView, Image,
  TouchableOpacity, Alert, StyleSheet, StatusBar,
  Linking, Modal
} from "react-native";

import AsyncStorage from "@react-native-async-storage/async-storage";
import * as ImagePicker from "expo-image-picker";

const BLUE = "#1769C2";

const CATEGORIES = [
  "T-shirts", "Pantalons", "Chaussures", "Sacs",
  "Montres", "Casquettes", "Lunettes", "Ceintures",
  "Pulls", "Robes", "Ensembles", "Sous-vêtements"
];

const CONTACTS = {
  whatsapp: "+224 626 00 46 90",
  phone: "+224 669 54 99 05",
  address: "Madina, Centre Commercial Diané, Conakry, Guinée"
};

// Identifiants provisoires de démonstration.
// À remplacer avant une utilisation réelle.
const ADMIN_PASSWORD = "DJUL2026";
const ADMIN_PIN = "2240";

export default function App() {
  const [tab, setTab] = useState("Accueil");
  const [dark, setDark] = useState(false);

  const [products, setProducts] = useState([]);
  const [cart, setCart] = useState([]);
  const [favorites, setFavorites] = useState([]);
  const [orders, setOrders] = useState([]);

  const [admin, setAdmin] = useState(false);
  const [login, setLogin] = useState(false);
  const [password, setPassword] = useState("");
  const [pin, setPin] = useState("");

  const [name, setName] = useState("");
  const [category, setCategory] = useState("T-shirts");
  const [price, setPrice] = useState("");
  const [promo, setPromo] = useState("");
  const [photo, setPhoto] = useState("");
  const [available, setAvailable] = useState(true);

  const [search, setSearch] = useState("");
  const [customer, setCustomer] = useState("");
  const [phone, setPhone] = useState("");
  const [address, setAddress] = useState("");
  const [payment, setPayment] = useState("Orange Money");

  const [trackCode, setTrackCode] = useState("");
  const [trackPhone, setTrackPhone] = useState("");

  const bg = dark ? "#141820" : "#F3F6FB";
  const card = dark ? "#252D39" : "#FFFFFF";
  const color = dark ? "#FFFFFF" : "#172033";
  const muted = dark ? "#B9C3D0" : "#697586";

  useEffect(() => {
    loadData();
  }, []);

  useEffect(() => {
    saveData();
  }, [products, orders, favorites]);

  async function loadData() {
    try {
      const data = await AsyncStorage.multiGet([
        "djul_products",
        "djul_orders",
        "djul_favorites"
      ]);

      const saved = Object.fromEntries(data);

      if (saved.djul_products)
        setProducts(JSON.parse(saved.djul_products));

      if (saved.djul_orders)
        setOrders(JSON.parse(saved.djul_orders));

      if (saved.djul_favorites)
        setFavorites(JSON.parse(saved.djul_favorites));

    } catch (e) {
      console.log("Erreur de chargement", e);
    }
  }

  async function saveData() {
    try {
      await AsyncStorage.multiSet([
        ["djul_products", JSON.stringify(products)],
        ["djul_orders", JSON.stringify(orders)],
        ["djul_favorites", JSON.stringify(favorites)]
      ]);
    } catch (e) {
      console.log("Erreur de sauvegarde", e);
    }
  }

  const money = value =>
    Number(value || 0).toLocaleString("fr-FR") + " GNF";

  const finalPrice = p =>
    p.promo > 0 && p.promo < p.price ? p.promo : p.price;

  async function choosePhoto() {
    const permission =
      await ImagePicker.requestMediaLibraryPermissionsAsync();

    if (!permission.granted) {
      Alert.alert("Autorisation", "Autorise l'accès à tes photos.");
      return;
    }

    const result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      quality: 0.8
    });

    if (!result.canceled) {
      setPhoto(result.assets[0].uri);
    }
  }

  function addProduct() {
    if (!name.trim() || !price || !photo) {
      Alert.alert(
        "Informations manquantes",
        "Ajoute une photo, un nom et un prix."
      );
      return;
    }

    if (Number(price) <= 0) {
      Alert.alert("Prix incorrect", "Entre un prix valide.");
      return;
    }

    const product = {
      id: Date.now().toString(),
      name: name.trim(),
      category,
      price: Number(price),
      promo: Number(promo) || 0,
      photo,
      available
    };

    setProducts(old => [product, ...old]);

    setName("");
    setPrice("");
    setPromo("");
    setPhoto("");
    setAvailable(true);

    Alert.alert("DJUL SHOP", "Produit ajouté avec succès !");
  }

  function deleteProduct(id) {
    Alert.alert("Suppression", "Supprimer ce produit ?", [
      { text: "Annuler" },
      {
        text: "Supprimer",
        onPress: () =>
          setProducts(old => old.filter(p => p.id !== id))
      }
    ]);
  }

  function toggleAvailability(id) {
    setProducts(old => old.map(p =>
      p.id === id ? { ...p, available: !p.available } : p
    ));
  }

  function addToCart(product) {
    if (!product.available) {
      Alert.alert("Indisponible", "Ce produit est épuisé.");
      return;
    }

    setCart(old => [...old, product]);
    Alert.alert("Panier", "Produit ajouté !");
  }

  function toggleFavorite(id) {
    setFavorites(old =>
      old.includes(id)
        ? old.filter(x => x !== id)
        : [...old, id]
    );
  }

  function placeOrder() {
    if (!customer.trim() || !phone.trim() || !address.trim()) {
      Alert.alert(
        "Commande",
        "Remplis ton nom, ton téléphone et ton adresse."
      );
      return;
    }

    if (!cart.length) {
      Alert.alert("Panier vide", "Ajoute des produits.");
      return;
    }

    const total = cart.reduce(
      (sum, p) => sum + finalPrice(p), 0
    );

    const code = "DJUL" + Date.now().toString().slice(-7);

    const order = {
      id: code,
      customer,
      phone,
      address,
      payment,
      products: [...cart],
      total,
      status: "En attente",
      date: new Date().toLocaleDateString("fr-FR")
    };

    setOrders(old => [order, ...old]);
    setCart([]);

    setCustomer("");
    setPhone("");
    setAddress("");

    setTrackCode(code);
    setTrackPhone(phone);

    setTab("Suivi");

    Alert.alert(
      "Commande enregistrée",
      "Ton numéro de commande : " + code
    );
  }

  function loginAdmin() {
    if (
      password === ADMIN_PASSWORD &&
      pin === ADMIN_PIN
    ) {
      setAdmin(true);
      setLogin(false);
      setPassword("");
      setPin("");
      setTab("Administration");
    } else {
      Alert.alert("Accès refusé", "Identifiants incorrects.");
    }
  }

  function changeStatus(id, status) {
    setOrders(old => old.map(o =>
      o.id === id ? { ...o, status } : o
    ));
  }

  function openWhatsApp() {
    Linking.openURL("https://wa.me/224626004690");
  }

  function callShop() {
    Linking.openURL("tel:+224669549905");
  }

  function openAddress() {
    const url =
      "https://www.google.com/maps/search/?api=1&query=" +
      encodeURIComponent(CONTACTS.address);

    Linking.openURL(url);
  }

  const filteredProducts = products.filter(p =>
    p.name.toLowerCase().includes(search.toLowerCase()) ||
    p.category.toLowerCase().includes(search.toLowerCase())
  );

  function header() {
    return (
      <View style={s.header}>
        <Text style={s.logo}>🛍️ DJUL SHOP</Text>
        <Text style={s.slogan}>
          MODE • STYLE • ÉLÉGANCE
        </Text>
        <Text style={s.country}>🇬🇳 GUINÉE CONAKRY</Text>
      </View>
    );
  }

  function productCard(p) {
    return (
      <View
        key={p.id}
        style={[s.product, { backgroundColor: card }]}
      >
        <Image
          source={{ uri: p.photo }}
          style={s.productImage}
        />

        <TouchableOpacity
          style={s.heart}
          onPress={() => toggleFavorite(p.id)}
        >
          <Text style={{ fontSize: 21 }}>
            {favorites.includes(p.id) ? "❤️" : "♡"}
          </Text>
        </TouchableOpacity>

        <Text style={[s.productName, { color }]}>
          {p.name}
        </Text>

        <Text style={s.price}>
          {money(finalPrice(p))}
        </Text>

        {p.promo > 0 && p.promo < p.price && (
          <>
            <Text style={s.oldPrice}>
              {money(p.price)}
            </Text>
            <Text style={{ color: "#E53935", fontSize: 12 }}>
              -{Math.round((1 - p.promo / p.price) * 100)}%
            </Text>
          </>
        )}

        <Text style={{ color: muted, fontSize: 12 }}>
          {p.category}
        </Text>

        {!p.available && (
          <Text style={{ color: "#E53935" }}>
            Épuisé
          </Text>
        )}

        {admin ? (
          <>
            <TouchableOpacity
              style={s.secondaryButton}
              onPress={() => toggleAvailability(p.id)}
            >
              <Text style={s.blueText}>
                {p.available ? "Marquer épuisé" : "Rendre disponible"}
              </Text>
            </TouchableOpacity>

            <TouchableOpacity
              style={s.deleteButton}
              onPress={() => deleteProduct(p.id)}
            >
              <Text style={s.white}>Supprimer</Text>
            </TouchableOpacity>
          </>
        ) : (
          <TouchableOpacity
            style={s.mainButton}
            onPress={() => addToCart(p)}
          >
            <Text style={s.white}>Ajouter au panier</Text>
          </TouchableOpacity>
        )}
      </View>
    );
  }

  function homePage() {
    return (
      <>
        {header()}

        <Text style={[s.heading, { color }]}>
          Bienvenue chez DJUL SHOP
        </Text>

        <Text style={{ color: muted }}>
          Mode homme et femme. Découvrez nos collections.
        </Text>

        <TextInput
          placeholder="Rechercher un vêtement..."
          placeholderTextColor={muted}
          value={search}
          onChangeText={setSearch}
          style={[s.input, { backgroundColor: card, color }]}
        />

        <Text style={[s.heading, { color }]}>
          Catégories
        </Text>

        <ScrollView horizontal>
          {CATEGORIES.map(c => (
            <TouchableOpacity
              key={c}
              style={s.category}
              onPress={() => {
                setSearch(c);
                setTab("Catégories");
              }}
            >
              <Text style={s.categoryText}>{c}</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>

        <Text style={[s.heading, { color }]}>
          Nos vêtements
        </Text>

        <View style={s.grid}>
          {filteredProducts.map(productCard)}
        </View>

        {!products.length && (
          <Text style={{ color: muted, textAlign: "center", padding: 25 }}>
            Nos nouveaux produits arrivent bientôt !
          </Text>
        )}
      </>
    );
  }

  function cartPage() {
    const total = cart.reduce(
      (sum, p) => sum + finalPrice(p), 0
    );

    return (
      <>
        {header()}
        <Text style={[s.heading, { color }]}>
          Mon panier
        </Text>

        {cart.map((p, i) => (
          <View
            key={i}
            style={[s.stat, { backgroundColor: card }]}
          >
            <Text style={{ color }}>{p.name}</Text>
            <Text style={s.price}>{money(finalPrice(p))}</Text>

            <TouchableOpacity
              onPress={() =>
                setCart(old => old.filter((_, index) => index !== i))
              }
            >
              <Text style={{ color: "#E53935" }}>
                Retirer du panier
              </Text>
            </TouchableOpacity>
          </View>
        ))}

        <Text style={[s.heading, { color }]}>
          Total : {money(total)}
        </Text>

        <Text style={{ color: muted }}>
          Les frais de livraison sont payés directement au livreur.
        </Text>

        <TextInput
          placeholder="Nom complet"
          value={customer}
          onChangeText={setCustomer}
          style={s.input}
        />

        <TextInput
          placeholder="Numéro de téléphone"
          value={phone}
          onChangeText={setPhone}
          keyboardType="phone-pad"
          style={s.input}
        />

        <TextInput
          placeholder="Ville, quartier et adresse"
          value={address}
          onChangeText={setAddress}
          style={s.input}
        />

        <Text style={{ color }}>Moyen de paiement</Text>

        {["Orange Money", "MTN Mobile Money"].map(p => (
          <TouchableOpacity
            key={p}
            style={s.secondaryButton}
            onPress={() => setPayment(p)}
          >
            <Text style={s.blueText}>
              {payment === p ? "✓ " : ""}{p}
            </Text>
          </TouchableOpacity>
        ))}

        <TouchableOpacity
          style={s.mainButton}
          onPress={placeOrder}
        >
          <Text style={s.white}>Confirmer la commande</Text>
        </TouchableOpacity>
      </>
    );
  }

  function adminPage() {
    const revenue = orders
      .filter(o => o.status === "Livrée")
      .reduce((sum, o) => sum + o.total, 0);

    return (
      <>
        {header()}

        <Text style={[s.heading, { color }]}>
          Administration DJUL SHOP
        </Text>

        <View style={[s.stat, { backgroundColor: card }]}>
          <Text style={{ color: muted }}>Chiffre d'affaires livré</Text>
          <Text style={s.statNumber}>{money(revenue)}</Text>
          <Text style={{ color }}>Commandes : {orders.length}</Text>
          <Text style={{ color }}>Produits : {products.length}</Text>
        </View>

        <Text style={[s.heading, { color }]}>
          Ajouter un produit
        </Text>

        <TouchableOpacity
          style={s.photoButton}
          onPress={choosePhoto}
        >
          {photo ? (
            <Image source={{ uri: photo }} style={s.preview}/>
          ) : (
            <Text style={s.white}>📸 Choisir une photo</Text>
          )}
        </TouchableOpacity>

        <TextInput
          placeholder="Nom du vêtement"
          value={name}
          onChangeText={setName}
          style={s.input}
        />

        <Text style={{ color }}>Catégorie</Text>

        <ScrollView horizontal>
          {CATEGORIES.map(c => (
            <TouchableOpacity
              key={c}
              onPress={() => setCategory(c)}
              style={[
                s.category,
                category === c && { backgroundColor: BLUE }
              ]}
            >
              <Text style={s.categoryText}>{c}</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>

        <TextInput
          placeholder="Prix en GNF"
          keyboardType="numeric"
          value={price}
          onChangeText={setPrice}
          style={s.input}
        />

        <TextInput
          placeholder="Prix promotionnel facultatif"
          keyboardType="numeric"
          value={promo}
          onChangeText={setPromo}
          style={s.input}
        />

        <TouchableOpacity
          style={s.secondaryButton}
          onPress={() => setAvailable(!available)}
        >
          <Text style={s.blueText}>
            Disponibilité : {available ? "Disponible" : "Épuisé"}
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={s.mainButton}
          onPress={addProduct}
        >
          <Text style={s.white}>+ Publier le produit</Text>
        </TouchableOpacity>

        <Text style={[s.heading, { color }]}>
          Mes commandes
        </Text>

        {orders.map(o => (
          <View
            key={o.id}
            style={[s.stat, { backgroundColor: card }]}
          >
            <Text style={{ color, fontWeight: "bold" }}>
              {o.id}
            </Text>
            <Text style={{ color }}>Client : {o.customer}</Text>
            <Text style={{ color }}>Téléphone : {o.phone}</Text>
            <Text style={{ color }}>Adresse : {o.address}</Text>
            <Text style={{ color }}>Paiement : {o.payment}</Text>
            <Text style={{ color }}>Total : {money(o.total)}</Text>
            <Text style={s.price}>Statut : {o.status}</Text>

            {[
              "Confirmée", "En préparation",
              "En livraison", "Livrée", "Refusée"
            ].map(st => (
              <TouchableOpacity
                key={st}
                style={s.secondaryButton}
                onPress={() => changeStatus(o.id, st)}
              >
                <Text style={s.blueText}>{st}</Text>
              </TouchableOpacity>
            ))}
          </View>
        ))}
      </>
    );
  }

  function trackingPage() {
    const found = orders.find(o =>
      o.id === trackCode.trim() &&
      o.phone === trackPhone.trim()
    );

    return (
      <>
        {header()}
        <Text style={[s.heading, { color }]}>
          Suivi de commande
        </Text>

        <TextInput
          placeholder="Numéro de commande"
          value={trackCode}
          onChangeText={setTrackCode}
          style={s.input}
        />

        <TextInput
          placeholder="Numéro de téléphone"
          value={trackPhone}
          onChangeText={setTrackPhone}
          style={s.input}
        />

        {found ? (
          <View style={[s.stat, { backgroundColor: card }]}>
            <Text style={{ color }}>Commande : {found.id}</Text>
            <Text style={s.statNumber}>{found.status}</Text>
            <Text style={{ color }}>Total : {money(found.total)}</Text>
          </View>
        ) : (
          <Text style={{ color: muted }}>
            Entre ton numéro de commande et ton téléphone.
          </Text>
        )}
      </>
    );
  }

  function contactPage() {
    return (
      <>
        {header()}
        <Text style={[s.heading, { color }]}>
          Contactez DJUL SHOP
        </Text>

        <TouchableOpacity
          style={s.whatsappButton}
          onPress={openWhatsApp}
        >
          <Text style={s.white}>
            WhatsApp : +224 626 00 46 90
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={s.mainButton}
          onPress={callShop}
        >
          <Text style={s.white}>
            Appeler : +224 669 54 99 05
          </Text>
        </TouchableOpacity>

        <Text style={[s.heading, { color }]}>
          Adresse de notre boutique
        </Text>

        <View style={[s.stat, { backgroundColor: card }]}>
          <Text style={{ color, fontWeight: "bold" }}>
            📍 Madina, Centre Commercial Diané
          </Text>
          <Text style={{ color, marginTop: 6 }}>
            Conakry, Guinée 🇬🇳
          </Text>
          <TouchableOpacity
            style={s.mainButton}
            onPress={openAddress}
          >
            <Text style={s.white}>Voir sur Google Maps</Text>
          </TouchableOpacity>
        </View>
      </>
    );
  }

  function profilePage() {
    return (
      <>
        {header()}
        <Text style={[s.heading, { color }]}>Mon profil</Text>

        {[
          ["❤️ Mes favoris", "Favoris"],
          ["🛒 Mon panier", "Panier"],
          ["📦 Suivre ma commande", "Suivi"],
          ["📞 Contacter DJUL SHOP", "Contact"]
        ].map(([label, destination]) => (
          <TouchableOpacity
            key={destination}
            style={s.secondaryButton}
            onPress={() => setTab(destination)}
          >
            <Text style={s.blueText}>{label}</Text>
          </TouchableOpacity>
        ))}

        <TouchableOpacity
          style={s.secondaryButton}
          onPress={() => setDark(!dark)}
        >
          <Text style={s.blueText}>
            {dark ? "☀️ Mode clair" : "🌙 Mode sombre"}
          </Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={s.secondaryButton}
          onPress={() => Alert.alert(
            "Conditions d'utilisation",
            "Les commandes sont soumises à confirmation par DJUL SHOP. Les frais de livraison sont payés au livreur."
          )}
        >
          <Text style={s.blueText}>Conditions d'utilisation</Text>
        </TouchableOpacity>

        <TouchableOpacity
          style={s.mainButton}
          onPress={() => setLogin(true)}
        >
          <Text style={s.white}>🔐 Administration</Text>
        </TouchableOpacity>
      </>
    );
  }

  return (
    <View style={{ flex: 1, backgroundColor: bg }}>
      <StatusBar barStyle={dark ? "light-content" : "dark-content"}/>

      <ScrollView contentContainerStyle={{ padding: 15, paddingBottom: 30 }}>
        {tab === "Accueil" && homePage()}
        {tab === "Catégories" && homePage()}
        {tab === "Panier" && cartPage()}
        {tab === "Profil" && profilePage()}
        {tab === "Contact" && contactPage()}
        {tab === "Suivi" && trackingPage()}

        {tab === "Favoris" && (
          <>
            {header()}
            <Text style={[s.heading, { color }]}>Mes favoris</Text>
            <View style={s.grid}>
              {products
                .filter(p => favorites.includes(p.id))
                .map(productCard)}
            </View>
          </>
        )}

        {tab === "Administration" && admin && adminPage()}
      </ScrollView>

      <View style={s.nav}>
        {[
          ["Accueil", "⌂"],
          ["Catégories", "▦"],
          ["Panier", "🛒"],
          ["Favoris", "♡"],
          ["Profil", "👤"]
        ].map(([label, icon]) => (
          <TouchableOpacity
            key={label}
            onPress={() => setTab(label)}
            style={{ alignItems: "center" }}
          >
            <Text style={{
              color: tab === label ? BLUE : "#777",
              fontSize: 21
            }}>{icon}</Text>
            <Text style={{
              color: tab === label ? BLUE : "#777",
              fontSize: 10
            }}>{label}</Text>
          </TouchableOpacity>
        ))}
      </View>

      <Modal visible={login} transparent animationType="slide">
        <View style={s.overlay}>
          <View style={s.loginBox}>
            <Text style={s.heading}>Espace administrateur</Text>

            <TextInput
              placeholder="Mot de passe"
              secureTextEntry
              value={password}
              onChangeText={setPassword}
              style={s.input}
            />

            <TextInput
              placeholder="Code PIN"
              secureTextEntry
              keyboardType="numeric"
              value={pin}
              onChangeText={setPin}
              style={s.input}
            />

            <TouchableOpacity
              style={s.mainButton}
              onPress={loginAdmin}
            >
              <Text style={s.white}>Se connecter</Text>
            </TouchableOpacity>

            <TouchableOpacity onPress={() => setLogin(false)}>
              <Text style={{ textAlign: "center", margin: 12 }}>
                Fermer
              </Text>
            </TouchableOpacity>
          </View>
        </View>
      </Modal>
    </View>
  );
}

const s = StyleSheet.create({
  header: {
    backgroundColor: BLUE,
    padding: 23,
    borderRadius: 17,
    marginBottom: 18
  },
  logo: {
    color: "#FFFFFF",
    fontSize: 28,
    fontWeight: "900",
    textAlign: "center"
  },
  slogan: {
    color: "#FFFFFF",
    textAlign: "center",
    marginTop: 6
  },
  country: {
    color: "#FFFFFF",
    textAlign: "center",
    marginTop: 8,
    fontSize: 11
  },
  heading: {
    fontSize: 20,
    fontWeight: "bold",
    marginVertical: 13
  },
  input: {
    backgroundColor: "#FFFFFF",
    padding: 14,
    borderRadius: 11,
    marginVertical: 7,
    color: "#172033",
    borderWidth: 1,
    borderColor: "#DCE5F0"
  },
  category: {
    backgroundColor: "#E6F0FC",
    padding: 12,
    borderRadius: 20,
    marginRight: 8,
    marginVertical: 8
  },
  categoryText: {
    color: BLUE,
